    92→function isRecord(v: unknown): v is Record<string, unknown> {
    93→  return typeof v === 'object' && v !== null
    94→}
    95→
    96→function isDeltaEvent(v: unknown): v is { delta: string } {
    97→  return isRecord(v) && typeof v.delta === 'string'
    98→}
    99→
   100→function isDoneEvent(v: unknown): v is { done: boolean } {
   101→  return isRecord(v) && typeof v.done === 'boolean'
   102→}
   103→
   104→function isErrorEvent(v: unknown): v is { error: string } {
   105→  return isRecord(v) && typeof v.error === 'string'
   106→}
   107→
   108→function isChatMessage(v: unknown): v is ChatMessage {
   109→  return (
   110→    isRecord(v) &&
   111→    typeof v.id === 'string' &&
   112→    (v.role === 'user' || v.role === 'assistant') &&
   113→    typeof v.content === 'string' &&
   114→    typeof v.createdAt === 'string'
   115→  )
   116→}
   117→
   118→function isHistoryResponse(v: unknown): v is { messages: ChatMessage[] } {
   119→  return isRecord(v) && Array.isArray(v.messages) && v.messages.every(isChatMessage)
   120→}
   121→
   122→function isSubjectId(v: string): v is SubjectId {
   123→  return SUBJECT_IDS.includes(v as SubjectId)
   124→}

   126→function uuid(): string {
   127→  if (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function') {
   128→    return crypto.randomUUID()
   129→  }
   130→  return `local-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 10)}`
   131→}
   132→
   133→const timeFormatter = new Intl.DateTimeFormat('zh-CN', {
   134→  hour: '2-digit',
   135→  minute: '2-digit',
   136→  hour12: false,
   137→})
   138→
   139→function formatTime(iso: string): string {
   140→  const d = new Date(iso)
   141→  return Number.isNaN(d.getTime()) ? '' : timeFormatter.format(d)
   142→}

   324→export function AssistantView() {
   325→  const assistantContext = useAppStore((s) => s.assistantContext)
   326→  const setAssistantContext = useAppStore((s) => s.setAssistantContext)
   327→  const { toast } = useToast()
   328→
   329→  const [sessionId, setSessionId] = useState<string | null>(null)
   330→  const [historyLoaded, setHistoryLoaded] = useState(false)
   331→  const [hadHistory, setHadHistory] = useState(false)
   332→  const [messages, setMessages] = useState<DisplayMessage[]>([])
   333→  const [input, setInput] = useState('')
   334→  const [generating, setGenerating] = useState(false)
   335→  const [streamingId, setStreamingId] = useState<string | null>(null)
   336→  const [atBottom, setAtBottom] = useState(true)
   337→  const [copiedId, setCopiedId] = useState<string | null>(null)
   338→  const [subjectFilter, setSubjectFilter] = useState<SubjectId | 'all'>('all')
   339→
   340→  const scrollRef = useRef<HTMLDivElement>(null)
   341→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   342→  const abortRef = useRef<AbortController | null>(null)
   343→  const stickRef = useRef(true)
   344→  const rafRef = useRef<number | null>(null)
   345→  const pendingRef = useRef('')
   346→  const streamIdRef = useRef<string | null>(null)
   347→  const sessionIdRef = useRef<string | null>(null)
   348→
   349→  /* ----- 会话初始化 + 历史加载 ----- */
   350→  useEffect(() => {
   351→    let sid: string | null = null
   352→    try {
   353→      sid = window.localStorage.getItem(SESSION_KEY)
   354→      if (!sid) {
   355→        sid = uuid()
   356→        window.localStorage.setItem(SESSION_KEY, sid)
   357→      }
   358→    } catch {
   359→      // 隐私模式等 localStorage 不可用时退化为内存会话
   360→      sid = uuid()
   361→    }
   362→    sessionIdRef.current = sid
   363→    setSessionId(sid)
   364→
   365→    let cancelled = false
   366→    fetch(`/api/chat?sessionId=${encodeURIComponent(sid)}`)
   367→      .then((r) => {
   368→        if (!r.ok) throw new Error(`HTTP ${r.status}`)
   369→        return r.json() as Promise<unknown>
   370→      })
   371→      .then((data: unknown) => {
   372→        if (cancelled) return
   373→        if (isHistoryResponse(data)) {
   374→          setMessages(data.messages)
   375→          setHadHistory(data.messages.length > 0)
   376→        }
   377→      })
   378→      .catch(() => {
   379→        if (!cancelled) {
   380→          toast({
   381→            title: '加载历史对话失败',
   382→            description: '请刷新页面重试',
   383→            variant: 'destructive',
   384→          })
   385→        }
   386→      })
   387→      .finally(() => {
   388→        if (!cancelled) setHistoryLoaded(true)
   389→      })
   390→
   391→    return () => {
   392→      cancelled = true
   393→    }
   394→  }, [toast])
   395→
   396→  /* ----- 卸载时中止请求 ----- */
   397→  useEffect(() => {
   398→    return () => {
   399→      abortRef.current?.abort()
   400→      if (rafRef.current != null) cancelAnimationFrame(rafRef.current)
   401→    }
   402→  }, [])
   403→
   404→  /* ----- 流式增量 rAF 批量渲染 ----- */
   405→  const flush = useCallback(() => {
   406→    const chunk = pendingRef.current
   407→    pendingRef.current = ''
   408→    const id = streamIdRef.current
   409→    if (!chunk || !id) return
   410→    setMessages((prev) =>
   411→      prev.map((m) => (m.id === id ? { ...m, content: m.content + chunk } : m))
   412→    )
   413→  }, [])
   414→
   415→  const scheduleFlush = useCallback(() => {
   416→    if (rafRef.current == null) {
   417→      rafRef.current = requestAnimationFrame(() => {
   418→        rafRef.current = null
   419→        flush()
   420→      })
   421→    }
   422→  }, [flush])
   423→
   424→  /* ----- 滚动控制：仅当接近底部时自动跟随 ----- */
   425→  const handleScroll = useCallback(() => {
   426→    const el = scrollRef.current
   427→    if (!el) return
   428→    const near = el.scrollHeight - el.scrollTop - el.clientHeight < 80
   429→    stickRef.current = near
   430→    setAtBottom((prev) => (prev === near ? prev : near))
   431→  }, [])
   432→
   433→  const scrollToBottom = useCallback((smooth: boolean) => {
   434→    const el = scrollRef.current
   435→    if (!el) return
   436→    stickRef.current = true
   437→    setAtBottom(true)
   438→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' })
   439→  }, [])
   440→
   441→  useEffect(() => {
   442→    if (!stickRef.current) return
   443→    const el = scrollRef.current
   444→    if (el) el.scrollTop = el.scrollHeight
   445→  }, [messages])
   446→
   447→  /* ----- 输入框自适应高度 ----- */
   448→  useEffect(() => {
   449→    const el = textareaRef.current
   450→    if (!el) return
   451→    el.style.height = 'auto'
   452→    el.style.height = `${Math.min(el.scrollHeight, MAX_INPUT_HEIGHT)}px`
   453→  }, [input])
   454→
   455→  /* ----- 发送 / 流式接收 ----- */
   456→  const runCompletion = useCallback(
   457→    async (opts: {
   458→      text: string
   459→      context?: ChatContext
   460→      appendUser: boolean
   461→      dropTrailingAssistant?: boolean
   462→    }) => {
   463→      const sid = sessionIdRef.current
   464→      const trimmed = opts.text.trim()
   465→      if (!sid || !trimmed) return
   466→
   467→      const aiId = uuid()
   468→      const now = new Date().toISOString()
   469→      const assistantMsg: DisplayMessage = {
   470→        id: aiId,
   471→        role: 'assistant',
   472→        content: '',
   473→        createdAt: now,
   474→        context: opts.context,
   475→      }
   476→      const userMsg: DisplayMessage = {
   477→        id: uuid(),
   478→        role: 'user',
   479→        content: trimmed,
   480→        createdAt: now,
   481→        context: opts.context,
   482→      }
   483→
   484→      setMessages((prev) => {
   485→        let base = prev
   486→        if (opts.dropTrailingAssistant) {
   487→          while (base.length > 0 && base[base.length - 1].role === 'assistant') {
   488→            base = base.slice(0, -1)
   489→          }
   490→        }
   491→        return opts.appendUser ? [...base, userMsg, assistantMsg] : [...base, assistantMsg]
   492→      })
   493→      setGenerating(true)
   494→      setStreamingId(aiId)
   495→      streamIdRef.current = aiId
   496→      pendingRef.current = ''
   497→      stickRef.current = true
   498→
   499→      const controller = new AbortController()
   500→      abortRef.current = controller
   501→
   502→      let streamError: string | null = null
   503→      let userAborted = false
   504→
   505→      try {
   506→        const res = await fetch('/api/chat', {
   507→          method: 'POST',
   508→          headers: { 'Content-Type': 'application/json' },
   509→          body: JSON.stringify({
   510→            sessionId: sid,
   511→            message: trimmed,
   512→            context: opts.context ?? undefined,
   513→          }),
   514→          signal: controller.signal,
   515→        })
   516→
   517→        if (!res.ok) {
   518→          streamError = `服务响应异常（HTTP ${res.status}）`
   519→        } else {
   520→          const body = res.body
   521→          if (!body) {
   522→            streamError = '响应流不可用'
   523→          } else {
   524→            const reader = body.getReader()
   525→            const decoder = new TextDecoder()
   526→            let buffer = ''
   527→            for (;;) {
   528→              const { done, value } = await reader.read()
   529→              if (done) break
   530→              buffer += decoder.decode(value, { stream: true })
   531→              const lines = buffer.split('\n')
   532→              buffer = lines.pop() ?? ''
   533→              for (const rawLine of lines) {
   534→                const line = rawLine.trim()
   535→                if (!line.startsWith('data:')) continue
   536→                const payload = line.slice(5).trim()
   537→                if (!payload || payload === '[DONE]') continue
   538→                let parsed: unknown
   539→                try {
   540→                  parsed = JSON.parse(payload)
   541→                } catch {
   542→                  continue // 忽略无法解析的行
   543→                }
   544→                if (isErrorEvent(parsed)) {
   545→                  streamError = parsed.error
   546→                  break
   547→                }
   548→                if (isDeltaEvent(parsed)) {
   549→                  pendingRef.current += parsed.delta
   550→                  scheduleFlush()
   551→                }
   552→                // { done: true } 为正常结束标记，无需处理
   553→              }
   554→              if (streamError) break
   555→            }
   556→            // 提前退出时释放底层连接
   557→            void reader.cancel().catch(() => {})
   558→          }
   559→        }
   560→      } catch (e) {
   561→        if (controller.signal.aborted) {
   562→          userAborted = true
   563→        } else if (e instanceof TypeError) {
   564→          streamError = '网络连接失败，请检查网络后重试'
   565→        } else {
   566→          streamError = e instanceof Error ? e.message : '请求失败，请重试'
   567→        }
   568→      } finally {
   569→        if (rafRef.current != null) {
   570→          cancelAnimationFrame(rafRef.current)
   571→          rafRef.current = null
   572→        }
   573→        flush()
   574→        abortRef.current = null
   575→        streamIdRef.current = null
   576→        setStreamingId(null)
   577→        setGenerating(false)
   578→      }
   579→
   580→      if (streamError !== null) {
   581→        const err = streamError
   582→        setMessages((prev) => prev.map((m) => (m.id === aiId ? { ...m, error: err } : m)))
   583→        toast({ title: '回复生成失败', description: err, variant: 'destructive' })
   584→      } else if (userAborted) {
   585→        setMessages((prev) => prev.map((m) => (m.id === aiId ? { ...m, stopped: true } : m)))
   586→      }
   587→    },
   588→    [flush, scheduleFlush, toast]
   589→  )
   590→
   591→  const send = useCallback(
   592→    (raw: string) => {
   593→      const text = raw.trim()
   594→      if (!text || generating || !historyLoaded || sessionId === null) return
   595→      setInput('')
   596→      const ctx: ChatContext | undefined = assistantContext ?? undefined
   597→      textareaRef.current?.focus()
   598→      void runCompletion({ text, context: ctx, appendUser: true })
   599→    },
   600→    [assistantContext, generating, historyLoaded, runCompletion, sessionId]
   601→  )
   602→
   603→  const retryLast = useCallback(() => {
   604→    if (generating || !historyLoaded) return
   605→    const lastUser = [...messages].reverse().find((m) => m.role === 'user')
   606→    if (!lastUser) return
   607→    void runCompletion({
   608→      text: lastUser.content,
   609→      context: lastUser.context ?? undefined,
   610→      appendUser: false,
   611→      dropTrailingAssistant: true,
   612→    })
   613→  }, [generating, historyLoaded, messages, runCompletion])
   614→
   615→  const stop = useCallback(() => {
   616→    abortRef.current?.abort()
   617→  }, [])
   618→
   619→  const clearConversation = useCallback(async () => {
   620→    const sid = sessionIdRef.current
   621→    if (!sid) return
   622→    abortRef.current?.abort()
   623→    try {
   624→      const res = await fetch(`/api/chat?sessionId=${encodeURIComponent(sid)}`, {
   625→        method: 'DELETE',
   626→      })
   627→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   628→      setMessages([])
   629→      setHadHistory(false)
   630→      toast({ title: '对话已清空', description: '已开始新的会话' })
   631→    } catch (e) {
   632→      const desc = e instanceof Error ? e.message : '请稍后重试'
   633→      toast({ title: '清空对话失败', description: desc, variant: 'destructive' })
   634→    }
   635→  }, [toast])
   636→
   637→  const copyMessage = useCallback(
   638→    async (message: DisplayMessage) => {
   639→      try {
   640→        await navigator.clipboard.writeText(message.content)
   641→        setCopiedId(message.id)
   642→        window.setTimeout(() => setCopiedId((cur) => (cur === message.id ? null : cur)), 1600)
   643→      } catch {
   644→        toast({
   645→          title: '复制失败',
   646→          description: '当前浏览器不支持剪贴板操作',
   647→          variant: 'destructive',
   648→        })
   649→      }
   650→    },
   651→    [toast]
   652→  )
   653→
   654→  /* ----- 输入事件 ----- */
   655→  const handleInputChange = useCallback((e: ChangeEvent<HTMLTextAreaElement>) => {
   656→    setInput(e.target.value)
   657→  }, [])
   658→
   659→  const handleKeyDown = useCallback(
   660→    (e: KeyboardEvent<HTMLTextAreaElement>) => {
   661→      if (e.key === 'Enter' && !e.shiftKey && !e.nativeEvent.isComposing) {
   662→        e.preventDefault()
   663→        send(input)
   664→      }
   665→    },
   666→    [input, send]
   667→  )
   668→
   669→  const handleFilterChange = useCallback((v: string) => {
   670→    setSubjectFilter(v === 'all' ? 'all' : isSubjectId(v) ? v : 'all')
   671→  }, [])
   672→
   673→  /* ----- 派生数据 ----- */
   674→  const contextInfo = useMemo(() => {
   675→    if (!assistantContext?.subjectId) return null
   676→    const subject = getSubject(assistantContext.subjectId)
   677→    const chapter = assistantContext.chapterId
   678→      ? getChapter(assistantContext.subjectId, assistantContext.chapterId)
   679→      : undefined
   680→    const parts: string[] = [subject?.name ?? '教材']
   681→    if (chapter) parts.push(`第 ${chapter.number} 章 ${chapter.title}`)
   682→    if (assistantContext.sectionTitle) parts.push(assistantContext.sectionTitle)
   683→    return {
   684→      label: parts.join(' · '),
   685→      theme: subject ? getSubjectTheme(subject.id) : null,
   686→    }
   687→  }, [assistantContext])
   688→
   689→  const filterSubject = useMemo(
   690→    () => (subjectFilter === 'all' ? null : getSubject(subjectFilter)),
   691→    [subjectFilter]
   692→  )
   693→
   694→  const canSend = input.trim().length > 0 && !generating && historyLoaded && sessionId !== null
   695→
   696→  const placeholder = !historyLoaded
   697→    ? '正在加载会话…'
   698→    : filterSubject
   699→      ? `聚焦「${filterSubject.name}」· 输入你的生物学问题…`
   700→      : '输入你的生物学问题，例如：解释糖酵解的反应历程…'
   701→
   702→  /* ----- 渲染 ----- */
   703→  return (
   704→    <div className="mx-auto flex h-[calc(100dvh-10rem)] min-h-[460px] w-full max-w-4xl flex-col px-3 pt-4 sm:px-4 lg:h-[calc(100dvh-8.5rem)]">
   705→      {/* 顶部：编辑式学术头部（含工具栏） */}
   706→      <header className="relative shrink-0 overflow-hidden rounded-t-2xl border bg-card">
   707→        <div
   708→          className="absolute inset-x-0 top-0 h-1 bg-gradient-to-r from-emerald-500 to-teal-600"
   709→          aria-hidden
   710→        />
   711→        <div className="px-4 pt-4 sm:px-5">
   712→          <div className="flex items-start justify-between gap-3">
   713→            <div className="min-w-0">
   714→              <p className="bio-eyebrow text-muted-foreground">AI Teaching Assistant</p>
   715→              <h1 className="mt-1.5 font-serif text-lg font-bold leading-snug tracking-tight sm:text-xl">
   716→                AI 智能助教
   717→              </h1>
   718→              <p className="mt-1 text-xs text-muted-foreground">
   719→                基于「101计划」教材体系的生物学专家助教
   720→              </p>
   721→            </div>
   722→            <span
   723→              className="hidden shrink-0 items-center gap-1.5 pt-1 text-[10px] text-muted-foreground sm:inline-flex"
   724→              aria-live="polite"
   725→            >
   726→              <span
   727→                className={cn(
   728→                  'h-1.5 w-1.5 rounded-full',
   729→                  generating ? 'animate-pulse bg-amber-500' : 'bg-emerald-500'
   730→                )}
   731→                aria-hidden
   732→              />
   733→              {generating ? '生成中' : '在线'}
   734→            </span>
   735→          </div>
   736→        </div>
   737→        <div className="bio-rule mt-3.5" aria-hidden />
   738→
   739→        {/* 工具栏：教材上下文 / 学科过滤 / 清空 */}
   740→        <div className="flex items-center gap-2 px-3 py-2 sm:px-4">
   741→          <div className="flex min-w-0 flex-1 items-center">
   742→            {contextInfo ? (
   743→              <Badge
   744→                variant="outline"
   745→                className={cn(
   746→                  'gap-1.5 py-1 pl-2.5 pr-1 text-[11px] font-medium',
   747→                  contextInfo.theme?.classes.badge ??
   748→                    'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/30'
   749→                )}
   750→              >
   751→                <BookOpenText className="h-3 w-3 shrink-0" aria-hidden />
   752→                <span className="max-w-[13rem] truncate sm:max-w-[24rem]" title={contextInfo.label}>
   753→                  {contextInfo.label}
   754→                </span>
   755→                <button
   756→                  type="button"
   757→                  onClick={() => setAssistantContext(null)}
   758→                  aria-label="清除教材上下文"
   759→                  className="ml-0.5 rounded-full p-0.5 transition-colors hover:bg-foreground/10"
   760→                >
   761→                  <X className="h-3 w-3" aria-hidden />
   762→                </button>
   763→              </Badge>
   764→            ) : filterSubject ? (
   765→              <Badge
   766→                variant="outline"
   767→                className="gap-1.5 border-emerald-500/30 bg-emerald-500/15 py-1 pl-2.5 pr-1 text-[11px] font-medium text-emerald-700 dark:text-emerald-400"
   768→              >
   769→                <Sparkles className="h-3 w-3 shrink-0" aria-hidden />
   770→                <span className="truncate">提问聚焦：{filterSubject.name}</span>
   771→                <button
   772→                  type="button"
   773→                  onClick={() => setSubjectFilter('all')}
   774→                  aria-label="取消学科聚焦"
   775→                  className="ml-0.5 rounded-full p-0.5 transition-colors hover:bg-foreground/10"
   776→                >
   777→                  <X className="h-3 w-3" aria-hidden />
   778→                </button>
   779→              </Badge>
   780→            ) : (
   781→              <p className="flex min-w-0 items-center gap-1.5 text-[11px] text-muted-foreground">
   782→                <Lightbulb className="h-3.5 w-3.5 shrink-0 text-muted-foreground/70" aria-hidden />
   783→                <span className="truncate">
   784→                  提示：从知识点阅读页点击「向 AI 提问」可携带教材上下文
   785→                </span>
   786→              </p>
   787→            )}
   788→          </div>
   789→
   790→          <div className="flex shrink-0 items-center gap-1.5">
   791→            <Select value={subjectFilter} onValueChange={handleFilterChange}>
   792→              <SelectTrigger
   793→                size="sm"
   794→                aria-label="选择学科聚焦范围"
   795→                className="h-8 w-[124px] rounded-lg text-xs text-muted-foreground"
   796→              >
   797→                <SelectValue />
   798→              </SelectTrigger>
   799→              <SelectContent>
   800→                <SelectItem value="all">全部学科</SelectItem>
   801→                {subjects.map((s) => (
   802→                  <SelectItem key={s.id} value={s.id}>
   803→                    {s.name}
   804→                  </SelectItem>
   805→                ))}
   806→              </SelectContent>
   807→            </Select>
   808→
   809→            <AlertDialog>
   810→              <AlertDialogTrigger asChild>
   811→                <Button
   812→                  type="button"
   813→                  variant="ghost"
   814→                  size="icon"
   815→                  disabled={messages.length === 0}
   816→                  aria-label="清空对话"
   817→                  className="h-8 w-8 rounded-lg text-muted-foreground hover:text-destructive"
   818→                >
   819→                  <Trash2 className="h-4 w-4" aria-hidden />
   820→                </Button>
   821→              </AlertDialogTrigger>
   822→              <AlertDialogContent className="max-w-sm">
   823→                <AlertDialogHeader>
   824→                  <AlertDialogTitle>清空当前对话？</AlertDialogTitle>
   825→                  <AlertDialogDescription>
   826→                    将删除本会话的全部 {messages.length} 条消息记录，此操作无法恢复。
   827→                  </AlertDialogDescription>
   828→                </AlertDialogHeader>
   829→                <AlertDialogFooter>
   830→                  <AlertDialogCancel>取消</AlertDialogCancel>
   831→                  <AlertDialogAction
   832→                    onClick={() => void clearConversation()}
   833→                    className="bg-destructive text-white hover:bg-destructive/90"
   834→                  >
   835→                    确认清空
   836→                  </AlertDialogAction>
   837→                </AlertDialogFooter>
   838→              </AlertDialogContent>
   839→            </AlertDialog>
   840→          </div>
   841→        </div>
   842→      </header>
   843→
   844→      {/* 消息区 */}
   845→      <div className="relative min-h-0 flex-1 border-x bg-background">
   846→        <div
   847→          ref={scrollRef}
   848→          onScroll={handleScroll}
   849→          role="log"
   850→          aria-label="对话消息"
   851→          aria-live="polite"
   852→          className="bio-scroll h-full overflow-y-auto px-3 py-4 sm:px-5"
   853→        >
   854→          {!historyLoaded ? (
   855→            <HistorySkeleton />
   856→          ) : messages.length === 0 ? (
   857→            <QuickQuestions onPick={send} />
   858→          ) : (
   859→            <div className="flex flex-col gap-5">
   860→              {hadHistory ? (
   861→                <div className="flex items-center gap-3" aria-hidden>
   862→                  <div className="bio-rule h-px flex-1" />
   863→                  <span className="text-[10px] text-muted-foreground">历史消息</span>
   864→                  <div className="h-px flex-1 bg-gradient-to-l from-border to-transparent" />
   865→                </div>
   866→              ) : null}
   867→              {messages.map((m) => (
   868→                <MessageBubble
   869→                  key={m.id}
   870→                  message={m}
   871→                  streaming={m.id === streamingId}
   872→                  copied={copiedId === m.id}
   873→                  onCopy={copyMessage}
   874→                  onRetry={retryLast}
   875→                />
   876→              ))}
   877→            </div>
   878→          )}
   879→          <div className="h-2" aria-hidden />
   880→        </div>
   881→
   882→        {!atBottom && (
   883→          <button
   884→            type="button"
   885→            onClick={() => scrollToBottom(true)}
   886→            aria-label="滚动到最新消息"
   887→            className="absolute bottom-3 right-3 flex h-9 w-9 items-center justify-center rounded-full border bg-background text-muted-foreground shadow-sm transition-colors hover:text-foreground"
   888→          >
   889→            <ChevronDown className="h-4 w-4" aria-hidden />
   890→          </button>
   891→        )}
   892→      </div>
   893→
   894→      {/* 输入区 */}
   895→      <div className="shrink-0 rounded-b-2xl border border-t bg-card p-2.5 pb-[max(0.625rem,env(safe-area-inset-bottom))] sm:p-3">
   896→        <div className="flex items-end gap-2 rounded-lg border bg-background px-2 py-1 transition-shadow focus-within:border-ring/50 focus-within:ring-2 focus-within:ring-ring/40">
   897→          <Textarea
   898→            ref={textareaRef}
   899→            value={input}
   900→            onChange={handleInputChange}
   901→            onKeyDown={handleKeyDown}
   902→            rows={1}
   903→            placeholder={placeholder}
   904→            aria-label="输入你的问题"
   905→            disabled={!historyLoaded}
   906→            className="min-h-[44px] flex-1 resize-none border-0 bg-transparent px-2 py-2.5 shadow-none focus-visible:border-transparent focus-visible:ring-0 focus-visible:ring-offset-0 dark:bg-transparent"
   907→          />
   908→          {generating ? (
   909→            <Button
   910→              type="button"
   911→              variant="outline"
   912→              size="icon"
   913→              onClick={stop}
   914→              aria-label="停止生成"
   915→              className="h-10 w-10 shrink-0 rounded-lg border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   916→            >
   917→              <Square className="h-4 w-4" aria-hidden />
   918→            </Button>
   919→          ) : (
   920→            <Button
   921→              type="button"
   922→              size="icon"
   923→              onClick={() => send(input)}
   924→              disabled={!canSend}
   925→              aria-label="发送问题"
   926→              className="h-10 w-10 shrink-0 rounded-lg"
   927→            >
   928→              <Send className="h-4 w-4" aria-hidden />
   929→            </Button>
   930→          )}
   931→        </div>
   932→        <p className="mt-1.5 flex flex-wrap items-center gap-x-2 gap-y-0.5 px-1 text-[10px] leading-relaxed text-muted-foreground">
   933→          <span>Enter 发送 · Shift + Enter 换行</span>
   934→          {filterSubject ? <span>当前聚焦「{filterSubject.name}」（提示性，不严格限定）</span> : null}
   935→          <span className="ml-auto hidden sm:inline">回答由 AI 生成，请注意核对关键数据</span>
   936→        </p>
   937→      </div>
   938→    </div>
   939→  )
   940→}
