    45→/** 相邻小节引用（用于上一节/下一节跨章导航） */
    46→interface SectionRef {
    47→  chapter: Chapter
    48→  section: Section
    49→}
    50→
    51→/** 计算上一节/下一节（跨章节边界时自动跳到上一章最后/下一章第一节） */
    52→function computeNeighbors(
    53→  subject: Subject | undefined,
    54→  chapter: Chapter | undefined,
    55→  chapterIndex: number,
    56→  sectionIndex: number
    57→): { prev: SectionRef | null; next: SectionRef | null } {
    58→  if (!subject || !chapter || sectionIndex < 0) {
    59→    return { prev: null, next: null }
    60→  }
    61→  let prev: SectionRef | null = null
    62→  let next: SectionRef | null = null
    63→  if (sectionIndex > 0) {
    64→    prev = { chapter, section: chapter.sections[sectionIndex - 1] }
    65→  } else if (chapterIndex > 0) {
    66→    const pc = subject.chapters[chapterIndex - 1]
    67→    if (pc.sections.length > 0) {
    68→      prev = { chapter: pc, section: pc.sections[pc.sections.length - 1] }
    69→    }
    70→  }
    71→  if (sectionIndex < chapter.sections.length - 1) {
    72→    next = { chapter, section: chapter.sections[sectionIndex + 1] }
    73→  } else if (chapterIndex < subject.chapters.length - 1) {
    74→    const nc = subject.chapters[chapterIndex + 1]
    75→    if (nc.sections.length > 0) {
    76→      next = { chapter: nc, section: nc.sections[0] }
    77→    }
    78→  }
    79→  return { prev, next }
    80→}

   523→// 章节目录树（左栏 / 移动端抽屉共用）

   525→function ChapterTree({
   526→  subject,
   527→  currentChapterId,
   528→  currentSectionId,
   529→  completedSections,
   530→  onSelect,
   531→}: {
   532→  subject: Subject
   533→  currentChapterId: string
   534→  currentSectionId: string
   535→  completedSections: string[]
   536→  onSelect: (chapterId: string, sectionId: string) => void
   537→}) {
   538→  const theme = getSubjectTheme(subject.id)
   539→  const [expandedId, setExpandedId] = useState<string | null>(currentChapterId)
   540→
   541→  // 当前章节变化时自动跟随展开
   542→  useEffect(() => {
   543→    setExpandedId(currentChapterId)
   544→  }, [currentChapterId])
   545→
   546→  return (
   547→    <nav aria-label={`${subject.name}章节目录`} className="space-y-1">
   548→      {subject.chapters.map((ch) => {
   549→        const expanded = expandedId === ch.id
   550→        const done = ch.sections.filter((s) => completedSections.includes(s.id)).length
   551→        const allDone = done === ch.sections.length && done > 0
   552→        return (
   553→          <div
   554→            key={ch.id}
   555→            className={cn(
   556→              'rounded-lg transition-colors',
   557→              expanded && theme.classes.bgSoft
   558→            )}
   559→          >
   560→            <button
   561→              onClick={() => setExpandedId(expanded ? null : ch.id)}
   562→              aria-expanded={expanded}
   563→              className="flex w-full items-center gap-2 rounded-lg px-2.5 py-2 text-left text-sm outline-none transition-colors hover:bg-accent focus-visible:ring-2 focus-visible:ring-ring"
   564→            >
   565→              <span
   566→                className={cn(
   567→                  'flex h-6 w-6 shrink-0 items-center justify-center rounded-md text-[11px] font-bold tabular-nums',
   568→                  allDone
   569→                    ? 'bg-primary text-primary-foreground'
   570→                    : cn(theme.classes.bgSoft, theme.classes.text)
   571→                )}
   572→                aria-hidden="true"
   573→              >
   574→                {allDone ? <CheckCircle2 className="h-3.5 w-3.5" /> : ch.number}
   575→              </span>
   576→              <span
   577→                className={cn(
   578→                  'min-w-0 flex-1 truncate',
   579→                  ch.id === currentChapterId ? 'font-semibold' : 'font-medium'
   580→                )}
   581→              >
   582→                {ch.title}
   583→              </span>
   584→              <span className="shrink-0 text-[10px] tabular-nums text-muted-foreground">
   585→                {done}/{ch.sections.length}
   586→              </span>
   587→              <ChevronRight
   588→                className={cn(
   589→                  'h-3.5 w-3.5 shrink-0 text-muted-foreground transition-transform duration-200',
   590→                  expanded && 'rotate-90'
   591→                )}
   592→                aria-hidden="true"
   593→              />
   594→            </button>
   595→            {expanded && (
   596→              <ul className="mt-0.5 space-y-0.5 pb-1.5 pl-3 pr-1.5">
   597→                {ch.sections.map((sec, i) => {
   598→                  const isCurrentSec = sec.id === currentSectionId
   599→                  const secDone = completedSections.includes(sec.id)
   600→                  return (
   601→                    <li key={sec.id}>
   602→                      <button
   603→                        onClick={() => onSelect(ch.id, sec.id)}
   604→                        aria-current={isCurrentSec ? 'page' : undefined}
   605→                        className={cn(
   606→                          'flex w-full items-center gap-2 rounded-md border-l-2 px-2 py-1.5 text-left text-xs transition-colors outline-none focus-visible:ring-2 focus-visible:ring-ring',
   607→                          isCurrentSec
   608→                            ? cn(
   609→                                'font-semibold',
   610→                                theme.classes.border,
   611→                                theme.classes.bgSoft,
   612→                                theme.classes.text
   613→                              )
   614→                            : 'border-transparent text-muted-foreground hover:bg-accent hover:text-foreground'
   615→                        )}
   616→                      >
   617→                        {secDone ? (
   618→                          <CheckCircle2
   619→                            className="h-3.5 w-3.5 shrink-0 text-emerald-500"
   620→                            aria-label="已完成"
   621→                          />
   622→                        ) : (
   623→                          <span
   624→                            className="flex h-3.5 w-3.5 shrink-0 items-center justify-center text-[10px] font-medium tabular-nums text-muted-foreground/60"
   625→                            aria-hidden="true"
   626→                          >
   627→                            {i + 1}
   628→                          </span>
   629→                        )}
   630→                        <span className="min-w-0 flex-1 truncate">
   631→                          {sec.title}
   632→                        </span>
   633→                      </button>
   634→                    </li>
   635→                  )
   636→                })}
   637→              </ul>
   638→            )}
   639→          </div>
   640→        )
   641→      })}
   642→    </nav>
   643→  )
   644→}

   647→// 右栏：本节要点 / 相关术语 / 学习操作（桌面右栏 + 移动端下方共用）
   648→// ============================================================
   649→function SectionAside({
   650→  subjectId,
   651→  section,
   652→  isCompleted,
   653→  pending,
   654→  onToggleComplete,
   655→  onAskAssistant,
   656→  onOpenGlossary,
   657→}: {
   658→  subjectId: SubjectId
   659→  section: Section
   660→  isCompleted: boolean
   661→  pending: boolean
   662→  onToggleComplete: () => void
   663→  onAskAssistant: () => void
   664→  onOpenGlossary: () => void
   665→}) {
   666→  const theme = getSubjectTheme(subjectId)
   667→
   668→  return (
   669→    <>
   670→      {/* 本节要点 */}
   671→      {section.keyPoints.length > 0 && (
   672→        <Card className={cn('gap-3 py-4', theme.classes.border)}>
   673→          <CardHeader className="px-4 pb-0">
   674→            <CardTitle
   675→              className={cn('flex items-center gap-2 text-sm', theme.classes.text)}
   676→            >
   677→              <Lightbulb className="h-4 w-4 shrink-0" aria-hidden="true" />
   678→              本节要点
   679→            </CardTitle>
   680→          </CardHeader>
   681→          <CardContent className="px-4">
   682→            <ul className="space-y-2.5">
   683→              {section.keyPoints.map((kp, i) => (
   684→                <li key={i} className="flex gap-2.5 text-xs leading-relaxed">
   685→                  <span
   686→                    className={cn(
   687→                      'mt-px flex h-5 w-5 shrink-0 items-center justify-center rounded-full text-[10px] font-bold tabular-nums',
   688→                      theme.classes.bgSoft,
   689→                      theme.classes.text
   690→                    )}
   691→                    aria-hidden="true"
   692→                  >
   693→                    {i + 1}
   694→                  </span>
   695→                  <span className="min-w-0 flex-1">{kp}</span>
   696→                </li>
   697→              ))}
   698→            </ul>
   699→          </CardContent>
   700→        </Card>
   701→      )}
   702→
   703→      {/* 相关术语 */}
   704→      {section.terms.length > 0 && (
   705→        <Card className="gap-3 py-4">
   706→          <CardHeader className="px-4 pb-0">
   707→            <CardTitle className="flex items-center gap-2 text-sm">
   708→              <BookMarked
   709→                className="h-4 w-4 shrink-0 text-primary"
   710→                aria-hidden="true"
   711→              />
   712→              相关术语
   713→            </CardTitle>
   714→          </CardHeader>
   715→          <CardContent className="px-4">
   716→            <div className="flex flex-wrap gap-1.5">
   717→              {section.terms.map((term) => (
   718→                <button
   719→                  key={term}
   720→                  onClick={onOpenGlossary}
   721→                  className={cn(
   722→                    'inline-flex items-center gap-1 rounded-md border px-2 py-1 text-xs transition-all outline-none hover:-translate-y-px focus-visible:ring-2 focus-visible:ring-ring',
   723→                    theme.classes.badge
   724→                  )}
   725→                  aria-label={`在术语词典中查看「${term}」`}
   726→                >
   727→                  <BookMarked className="h-3 w-3 opacity-60" aria-hidden="true" />
   728→                  {term}
   729→                </button>
   730→              ))}
   731→            </div>
   732→            <p className="mt-2.5 text-[10px] text-muted-foreground">
   733→              点击术语可跳转术语词典查看详细释义
   734→            </p>
   735→          </CardContent>
   736→        </Card>
   737→      )}
   738→
   739→      {/* 学习操作 */}
   740→      <Card className="gap-3 py-4">
   741→        <CardContent className="space-y-3 px-4">
   742→          {isCompleted && (
   743→            <p className="flex items-center justify-center gap-1.5 text-xs font-medium text-emerald-600 dark:text-emerald-400">
   744→              <CheckCircle2 className="h-4 w-4" aria-hidden="true" />
   745→              已完成本节学习
   746→            </p>
   747→          )}
   748→          {isCompleted ? (
   749→            <Button
   750→              variant="outline"
   751→              className="w-full gap-2"
   752→              onClick={onToggleComplete}
   753→              disabled={pending}
   754→            >
   755→              {pending ? (
   756→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   757→              ) : (
   758→                <Undo2 className="h-4 w-4" aria-hidden="true" />
   759→              )}
   760→              取消已学
   761→            </Button>
   762→          ) : (
   763→            <Button
   764→              className="w-full gap-2 bg-emerald-600 text-white hover:bg-emerald-700"
   765→              onClick={onToggleComplete}
   766→              disabled={pending}
   767→            >
   768→              {pending ? (
   769→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   770→              ) : (
   771→                <CheckCircle2 className="h-4 w-4" aria-hidden="true" />
   772→              )}
   773→              标记为已学
   774→            </Button>
   775→          )}
   776→          <Separator />
   777→          <Tooltip>
   778→            <TooltipTrigger asChild>
   779→              <Button
   780→                variant="outline"
   781→                className={cn('w-full gap-2', theme.classes.hover)}
   782→                onClick={onAskAssistant}
   783→              >
   784→                <MessageSquareText
   785→                  className="h-4 w-4"
   786→                  aria-hidden="true"
   787→                />
   788→                向 AI 助教提问本节
   789→              </Button>
   790→            </TooltipTrigger>
   791→            <TooltipContent>将携带本节学习上下文进入对话</TooltipContent>
   792→          </Tooltip>
   793→        </CardContent>
   794→      </Card>
   795→    </>
   796→  )
   797→}

   802→function SectionNavCard({
   803→  direction,
   804→  target,
   805→  currentChapterId,
   806→  hover,
   807→  onClick,

   809→  direction: 'prev' | 'next'

   811→  currentChapterId: string
   812→  hover: string

   821→      className={cn(
   822→        'group flex flex-1 items-center gap-3 rounded-xl border bg-card p-4 text-left outline-none transition-all duration-200 hover:-translate-y-0.5 hover:shadow-md focus-visible:ring-2 focus-visible:ring-ring',

   829→          className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-muted transition-colors group-hover:bg-accent"

   837→          className={cn(
   838→            'flex items-center gap-1.5 text-[11px] text-muted-foreground',
   839→            isNext && 'sm:justify-end'

   841→        >
   842→          {crossChapter && (
   843→            <Badge variant="outline" className="h-4 px-1.5 text-[10px]">
