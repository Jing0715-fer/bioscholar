    31→// ---------------- 文本宽度估算（用于自动定宽，防溢出） ----------------
    32→const CJK = /[\u2E80-\u9FFF\uF900-\uFAFF\u3000-\u303F\uFF00-\uFFEF]/
    33→const WIDE = /[A-Z0-9Ａ-Ｚ０-９]/
    34→const THIN = /[·,.:;'"()\[\]{}?!\/|ilj I1]/  // 注意含空格
    35→const SUBSUP = /[\u2070-\u209F\u00B2\u00B3\u00B9]/
    36→/** 估算字符串渲染宽度（px）。size 为字号。 */
    37→export function textW(s: string, size: number, weight = 400): number {
    38→  let w = 0
    39→  for (const ch of s) {
    40→    if (CJK.test(ch)) w += size
    41→    else if (SUBSUP.test(ch)) w += size * 0.55
    42→    else if (WIDE.test(ch)) w += size * 0.62
    43→    else if (THIN.test(ch)) w += size * 0.32
    44→    else if ('→←↔↑↓⇌≥≤≈×±'.includes(ch)) w += size * 0.85
    45→    else w += size * 0.52
    46→  }
    47→  return w * (weight >= 600 ? 1.03 : 1)
    48→}

    78→  private a(o: TextOpt = {}) {
    79→    const s = ` font-size="${o.size ?? 18}"`
    80→    return (
    81→      s +
    82→      (o.fill ? ` fill="${o.fill}"` : '') +
    83→      (o.anchor ? ` text-anchor="${o.anchor}"` : '') +
    84→      (o.weight && o.weight !== 400 ? ` font-weight="${o.weight}"` : '') +
    85→      (o.italic ? ` font-style="italic"` : '') +
    86→      (o.ls ? ` letter-spacing="${o.ls}"` : '') +
    87→      (o.opacity != null ? ` opacity="${o.opacity}"` : '')
    88→    )
    89→  }
    90→  private sh(o: ShapeOpt = {}) {
    91→    return (
    92→      (o.fill && o.fill !== 'none' ? ` fill="${o.fill}"` : ' fill="none"') +
    93→      (o.fillOp != null ? ` fill-opacity="${o.fillOp}"` : '') +
    94→      (o.stroke ? ` stroke="${o.stroke}"` : '') +
    95→      (o.sw != null ? ` stroke-width="${o.sw ?? 1.5}"` : '') +
    96→      (o.dash ? ` stroke-dasharray="${o.dash}"` : '') +
    97→      (o.opacity != null ? ` opacity="${o.opacity}"` : '')
    98→    )
    99→  }
   100→  private lh(o: LineOpt = {}) {
   101→    const mk = o.marker ? ` marker-end="url(#arr-${o.marker})"` : ''
   102→    const ms = o.markerStart ? ` marker-start="url(#arr-${o.markerStart})"` : ''
   103→    return (
   104→      ` stroke="${o.stroke ?? '#334155'}"` +
   105→      (o.sw != null ? ` stroke-width="${o.sw ?? 1.5}"` : '') +
   106→      (o.dash ? ` stroke-dasharray="${o.dash}"` : '') +
   107→      (o.opacity != null ? ` opacity="${o.opacity}"` : '') +
   108→      mk + ms
   109→    )
   110→  }

   112→  /** 文本（默认左对齐，y 为基线） */

   125→  /** 自动换行的文本块（按估算宽度断行），返回下一行基线 y */
   126→  wtext(x: number, y: number, s: string, o: TextOpt & { maxW?: number; lh?: number } = {}): number {
   127→    const size = o.size ?? 17
   128→    const maxW = o.maxW ?? 260
   129→    const lh = o.lh ?? size * 1.5
   130→    const lines: string[] = []
   131→    let cur = ''
   132→    for (const word of s.split(/(?<=[，。；：、！？）])|(?=[（])/)) {
   133→      if (cur && textW(cur + word, size) > maxW) { lines.push(cur); cur = word }
   134→      else cur += word
   135→    }
   136→    if (cur) lines.push(cur)
   137→    lines.forEach((ln, i) => this.text(x, y + i * lh, ln, o))
   138→    return y + lines.length * lh
   139→  }

   197→  /** 自动定宽标签盒（cx,cy 为盒心）。返回盒宽。tfill 为文字色（区别于盒填充 fill）。 */
   198→  tag(cx: number, cy: number, s: string, o: TextOpt & ShapeOpt & { pad?: number; minh?: number; tfill?: string } = {}): number {
   199→    const size = o.size ?? 17
   200→    const w = textW(s, size, o.weight ?? 400) + (o.pad ?? 16) * 2
   201→    const h = o.minh ?? size * 1.7
   202→    this.rect(cx - w / 2, cy - h / 2, w, h, {
   203→      fill: o.fill ?? C.accL, stroke: o.stroke, sw: o.sw ?? 1.5, rx: o.rx ?? 7, dash: o.dash, opacity: o.opacity,
   204→    })
   205→    this.text(cx, cy + size * 0.36, s, { size, fill: o.tfill ?? C.ink, weight: o.weight, anchor: 'middle' })
   206→    return w
   207→  }
   208→  /** 图例（色块 + 文字横排），返回总宽 */
   209→  legend(x: number, y: number, items: [string, string][], o: { size?: number; gap?: number; stroke?: boolean } = {}): number {
   210→    const size = o.size ?? 15
   211→    const gap = o.gap ?? 22
   212→    let cx = x
   213→    for (const [label, color] of items) {
   214→      this.rect(cx, y - size * 0.42, size * 1.1, size * 0.84, { fill: color, stroke: o.stroke === false ? undefined : C.line, sw: 1, rx: 3 })
   215→      this.text(cx + size * 1.1 + 8, y + size * 0.36, label, { size, fill: C.sub })
   216→      cx += size * 1.1 + 8 + textW(label, size) + gap
   217→    }
   218→    return cx - x - gap
   219→  }

   222→  /** 水平双链 DNA 示意：两条相位相反的正弦链 + 碱基横档 */
   223→  dna(x: number, y: number, w: number, o: { amp?: number; period?: number; stroke?: string; sw?: number; rung?: boolean; rungC?: string; opacity?: number } = {}): this {
   224→    const amp = o.amp ?? 9
   225→    const period = o.period ?? 46
   226→    const top: string[] = [`M${x},${y}`]
   227→    const bot: string[] = [`M${x},${y}`]
   228→    let phase = 0
   229→    for (let t = 0; t <= w; t += 2) {
   230→      const dy = amp * Math.sin((2 * Math.PI * t) / period)
   231→      if (t % 2 === 0) {
   232→        top.push(`L${(x + t).toFixed(1)},${(y - dy).toFixed(1)}`)
   233→        bot.push(`L${(x + t).toFixed(1)},${(y + dy).toFixed(1)}`)
   234→      }
   235→      phase = dy
   236→    }
   237→    const stroke = o.stroke ?? C.dna
   238→    this.els.push(`<path d="${top.join('')}" fill="none" stroke="${stroke}" stroke-width="${o.sw ?? 2.6}"${o.opacity != null ? ` opacity="${o.opacity}"` : ''}/>`)
   239→    this.els.push(`<path d="${bot.join('')}" fill="none" stroke="${stroke}" stroke-width="${o.sw ?? 2.6}"${o.opacity != null ? ` opacity="${o.opacity}"` : ''}/>`)
   240→    if (o.rung !== false) {
   241→      for (let t = period / 4; t < w; t += period / 2) {
   242→        const dy = amp * Math.sin((2 * Math.PI * t) / period)
   243→        this.line(x + t, y - dy, x + t, y + dy, { stroke: o.rungC ?? stroke, sw: 1.4, opacity: 0.55 })
   244→      }
   245→    }
   246→    void phase
   247→    return this
   248→  }
   249→  /** 基因簇轨道：在基线上排布基因方块并标注 */
   250→  genes(x: number, y: number, w: number, list: { label: string; frac: number; fill?: string; stroke?: string; h?: number; up?: boolean; size?: number }[]): this {
   251→    let cx = x
   252→    this.line(x, y, x + w, y, { stroke: C.faint, sw: 1.6 })
   253→    for (const g of list) {
   254→      const gw = w * g.frac
   255→      const h = g.h ?? 30
   256→      this.rect(cx + 1, y - h / 2, Math.max(gw - 2, 6), h, { fill: g.fill ?? C.dnaL, stroke: g.stroke ?? C.dna, sw: 1.6, rx: 4 })
   257→      const size = g.size ?? (gw > textW(g.label, 15) + 12 ? 15 : 13)
   258→      this.ctext(cx + gw / 2, g.up ? y - h / 2 - 12 : y + h / 2 + 18, g.label, { size, weight: 600, fill: C.sub })
   259→      cx += gw
   260→    }
   261→    return this
   262→  }
   263→  /** 泳道式凝胶图（孔在上、梯度带）。bands: [相对迁移(0顶-1底), 颜色, 标签?] */
   264→  gel(x: number, y: number, w: number, h: number, lanes: { label: string; bands: [number, string][] }[], o: { size?: number } = {}): this {
   265→    const size = o.size ?? 14
   266→    const n = lanes.length
   267→    const lw = (w - 40) / n
   268→    this.rect(x, y, w, h, { fill: '#ffffff', stroke: C.sub, sw: 2, rx: 4 })
   269→    lanes.forEach((lane, i) => {
   270→      const lx = x + 20 + i * lw
   271→      // 上样孔
   272→      this.rect(lx + lw * 0.18, y + 10, lw * 0.64, 7, { fill: C.ink, rx: 2 })
   273→      // 条带
   274→      for (const [rf, color] of lane.bands) {
   275→        const by = y + 26 + rf * (h - 40)
   276→        this.rect(lx + lw * 0.14, by - 4.5, lw * 0.72, 9, { fill: color, opacity: 0.9, rx: 2 })
   277→      }
   278→      this.ctext(lx + lw / 2, y + h + 20, lane.label, { size, fill: C.sub })
   279→    })
   280→    return this

   282→  /** 阶梯/台阶图（用于容量分级、范式演进等）。items 自动等分宽度。 */
   283→  stairs(x: number, y: number, w: number, h: number, items: string[], o: { fill?: string; stroke?: string; size?: number } = {}): this {
   284→    const n = items.length
   285→    const swStep = w / n
   286→    const shStep = h / n
   287→    items.forEach((s, i) => {
   288→      const bx = x + i * swStep
   289→      const by = y + h - (i + 1) * shStep
   290→      this.rect(bx, by, swStep - 6, shStep, { fill: o.fill ?? C.accL, stroke: o.stroke ?? C.acc, sw: 1.6, rx: 5 })
   291→      this.wtext(bx + 14, by + 24, s, { size: o.size ?? 15, fill: C.ink, maxW: swStep - 24, lh: 21 })
   292→    })
   293→    return this
   294→  }
   295→  /** 大括号标注（右侧），label 放在括号尖角处 */
   296→  brace(x: number, y1: number, y2: number, label: string, o: { size?: number; fill?: string; side?: 'left' | 'right' } = {}): this {
   297→    const mid = (y1 + y2) / 2
   298→    const d = y2 > y1 ? 14 : -14
   299→    const dir = o.side === 'left' ? -1 : 1
   300→    const p = `M${x + dir * 6},${y1} q${-dir * 12},0 ${-dir * 12},${d / 2} L${x + dir * 6},${mid - d / 3} q${-dir * 4},${d / 6} ${dir * 6},${d / 3} q${dir * 10},${d / 6} ${dir * 6},${d / 3} L${x + dir * 6},${y2} q${-dir * 12},0 ${-dir * 12},${-d / 2}`
   301→    this.path(p, { stroke: o.fill ?? C.mute, sw: 1.8 })
   302→    this.text(x + dir * 26, mid + 6, label, { size: o.size ?? 15, fill: o.fill ?? C.mute, anchor: o.side === 'left' ? 'end' : 'start' })
   303→    return this

   665→        const x2 = cx + (r + 12) * Math.cos(a); const y2 = cy + (r + 12) * Math.sin(a)
   666→        this.line(x1, y1, x2, y2, { stroke: st, sw: 2.4 })
   667→        this.circle(x2, y2, 4.6, { fill: st })
   668→      }
   669→      let d = ''
   670→      for (let i = 0; i <= 16; i++) {
   671→        const px = cx - r * 0.55 + (i * r * 1.1) / 16
   672→        const py = cy + r * 0.42 * Math.sin(i * 0.9)
   673→        d += `${i === 0 ? 'M' : 'L'} ${px.toFixed(1)},${py.toFixed(1)} `
   674→      }
   675→      this.path(d, { stroke: st, sw: 4.4, opacity: 0.85 })
   676→    } else if (shape === 'bullet') {
   677→      const bw = r * 1.1; const bh = r * 1.9
   678→      this.path(`M ${cx - bw / 2},${cy + bh / 2 - bw / 2} a ${bw / 2},${bw / 2} 0 0 1 ${bw},0 L ${cx + bw / 2},${cy - bh / 2 + bw * 0.35} q 0,${-bw * 0.35} ${-bw / 2},${-bw * 0.35} q ${-bw / 2},0 ${-bw / 2},${bw * 0.35} Z`, { fill, fillOp: 0.55, stroke: st, sw: 2.4 })
   679→      for (let i = 0; i < 9; i++) {
   680→        const a = (i / 9) * Math.PI * 1.4 - Math.PI * 0.2
   681→        const x1 = cx + (bw / 2) * Math.cos(a)
   682→        const y1 = cy + (bh / 2 - bw * 0.3) * Math.sin(a) + bh * 0.12
   683→        this.line(x1, y1, x1 + 11 * Math.cos(a), y1 + 11 * Math.sin(a), { stroke: st, sw: 2 })
   684→      }
   685→      let d = ''
   686→      for (let i = 0; i <= 12; i++) {
   687→        const px = cx - bw * 0.32 + (i * bw * 0.64) / 12
   688→        const py = cy + bh * 0.1 + bw * 0.28 * Math.sin(i * 1.1)
   689→        d += `${i === 0 ? 'M' : 'L'} ${px.toFixed(1)},${py.toFixed(1)} `
   690→      }
   691→      this.path(d, { stroke: st, sw: 4.2, opacity: 0.85 })
   692→    } else {
   693→      const bw = r * 0.9; const bh = r * 2.6
   694→      this.rect(cx - bw / 2, cy - bh / 2, bw, bh, { fill, fillOp: 0.6, stroke: st, sw: 2.2, rx: bw / 4 })
   695→      for (let i = 0; i < 9; i++) {
   696→        const py = cy - bh / 2 + 9 + (i * (bh - 18)) / 8
   697→        this.path(`M ${cx - bw / 2},${py.toFixed(1)} q ${bw / 4},${bw * 0.28} ${bw / 2},0 q ${bw / 4},${-bw * 0.28} ${bw / 2},0`, { stroke: st, sw: 2.4 })
   698→      }
   699→      this.line(cx, cy - bh / 2 - 9, cx, cy + bh / 2 + 9, { stroke: st, sw: 2.2, dash: '6 5' })
   700→    }
   701→    if (o.label) this.ctext(cx, cy + r * 2.4 + (shape === 'helical' ? r * 0.9 : 0) + 20, o.label, { size: 14.5, weight: 700, fill: st })
   702→    return this
   703→  }
   704→
   705→  /** 水平花括号 + 标注 */
   706→  braceH(x: number, y: number, w: number, o: { label?: string; flip?: boolean; fill?: string; size?: number } = {}): this {
   707→    const c = o.fill ?? C.mute
   708→    const dir = o.flip ? -1 : 1
   709→    const k = 9
   710→    this.path(`M ${x},${y + dir * k} Q ${x},${y} ${x + k},${y} M ${x + w - k},${y} Q ${x + w},${y} ${x + w},${y + dir * k}`, { stroke: c, sw: 1.7 })
   711→    this.line(x + k, y, x + w - k, y, { stroke: c, sw: 1.7 })
   712→    if (o.label) this.ctext(x + w / 2, y + dir * (k + 15), o.label, { size: o.size ?? 14, weight: 600, fill: c })
   713→    return this
   714→  }
   715→
   716→  /** 垂直花括号 + 标注 */
   717→  braceV(x: number, y: number, h: number, o: { label?: string; left?: boolean; fill?: string } = {}): this {
   718→    const c = o.fill ?? C.mute
   719→    const k = 9
   720→    const dir = o.left ? -1 : 1
   721→    this.path(`M ${x + dir * k},${y} Q ${x},${y} ${x},${y + k} M ${x},${y + h - k} Q ${x},${y + h} ${x + dir * k},${y + h}`, { stroke: c, sw: 1.7 })
   722→    this.line(x, y + k, x, y + h - k, { stroke: c, sw: 1.7 })
   723→    if (o.label) this.text(x + dir * (k + 9), y + h / 2, o.label, { size: 14, weight: 600, fill: c, anchor: o.left ? 'end' : 'start' })
   724→    return this
   725→  }
   726→}

   728→/** Catmull-Rom 平滑曲线 → cubic bezier path */
   729→function catmullRom(pts: [number, number][]): string {
   730→  if (pts.length < 2) return ''
   731→  if (pts.length === 2) return `M${pts[0][0]},${pts[0][1]} L${pts[1][0]},${pts[1][1]}`
   732→  let d = `M${pts[0][0]},${pts[0][1]}`
   733→  for (let i = 0; i < pts.length - 1; i++) {
   734→    const p0 = pts[Math.max(0, i - 1)]
   735→    const p1 = pts[i]
   736→    const p2 = pts[i + 1]
   737→    const p3 = pts[Math.min(pts.length - 1, i + 2)]
   738→    const c1x = p1[0] + (p2[0] - p0[0]) / 6
   739→    const c1y = p1[1] + (p2[1] - p0[1]) / 6
   740→    const c2x = p2[0] - (p3[0] - p1[0]) / 6
   741→    const c2y = p2[1] - (p3[1] - p1[1]) / 6
   742→    d += ` C${c1x.toFixed(1)},${c1y.toFixed(1)} ${c2x.toFixed(1)},${c2y.toFixed(1)} ${p2[0]},${p2[1]}`
   743→  }
   744→  return d
   745→}

   758→/** 按估算宽度把字符串断成 ≤maxLines 行（CJK 逐字断行，拉丁按词） */
   759→function wrapLines(s: string, size: number, maxW: number, maxLines: number): string[] {
   760→  const lines: string[] = []
   761→  let cur = ''
   762→  const tokens = s.split(/(?<=[，。；：、！？）,;:])\s*|(?<=[^，。；：、！？）,;:\s])\s+/)
   763→  for (const tk of tokens.length > 1 ? tokens : s.split('')) {
   764→    const cand = cur + tk
   765→    if (cur && textW(cand, size) > maxW) { lines.push(cur); cur = tk } else cur = cand
   766→  }
   767→  if (cur) lines.push(cur)
   768→  if (lines.length > maxLines) { // 压缩失败 → 均分
   769→    const per = Math.ceil(s.length / maxLines)
   770→    return Array.from({ length: maxLines }, (_, i) => s.slice(i * per, (i + 1) * per)).filter(Boolean)
   771→  }
   772→  return lines
   773→}

   775→export function scene(o: SceneOpt): string {
   776→  const w = o.w ?? 1400
   777→  const h = o.h ?? 1000
   778→  const head = o.head ?? 122
   779→  const b = new B()
   780→  o.draw(b)
   781→  const markers = (Object.keys(MARKER_FILL) as MarkerKey[])
   782→    .map(k => `<marker id="arr-${k}" viewBox="0 0 10 10" refX="8.5" refY="5" markerWidth="7" markerHeight="7" orient="auto-start-reverse"><path d="${MARKER_PATH}" fill="${MARKER_FILL[k]}"/></marker>`)
   783→    .join('\n    ')
   784→  // ---- 标题自适应：过宽则缩字号（下限 24）----
   785→  let tsize = 33
   786→  if (o.title) {
   787→    const tw = textW(o.title, tsize, 700)
   788→    if (tw > w - 100) tsize = Math.max(24, Math.floor(((w - 100) / tw) * tsize))
   789→  }
   790→  // ---- 副标题自适应：过宽优先两行（字号 17–19），仍溢出则缩字号（下限 13）----
   791→  let subLines: string[] = []
   792→  let ssize = 19
   793→  if (o.subtitle) {
   794→    const maxW = w - 90
   795→    if (textW(o.subtitle, ssize) <= maxW) subLines = [o.subtitle]
   796→    else {
   797→      ssize = 17
   798→      subLines = wrapLines(o.subtitle, ssize, maxW, 2)
   799→      while (subLines.some(ln => textW(ln, ssize) > maxW) && ssize > 13) {
   800→        ssize--
   801→        subLines = wrapLines(o.subtitle, ssize, maxW, 2)
   802→      }
   803→    }
   804→  }
   805→  const two = subLines.length > 1
   806→  const tBaseline = two ? head - 84 : head - 66
   807→  const title = o.title
   808→    ? `<text x="${w / 2}" y="${tBaseline}" text-anchor="middle" font-size="${tsize}" font-weight="700" fill="${C.ink}">${esc(o.title)}</text>`
   809→    : ''
   810→  const sub = subLines.length
   811→    ? (two
   812→        ? `<text x="${w / 2}" y="${head - 48}" text-anchor="middle" font-size="${ssize}" fill="${C.mute}">${esc(subLines[0])}</text>\n  <text x="${w / 2}" y="${head - 22}" text-anchor="middle" font-size="${ssize}" fill="${C.mute}">${esc(subLines[1] ?? '')}</text>`
   813→        : `<text x="${w / 2}" y="${head - 32}" text-anchor="middle" font-size="${ssize}" fill="${C.mute}">${esc(subLines[0])}</text>`)
   814→    : ''
   815→  return `<?xml version="1.0" encoding="UTF-8"?>
   816→<svg xmlns="http://www.w3.org/2000/svg" width="${w}" height="${h}" viewBox="0 0 ${w} ${h}" font-family="Noto Serif SC, LXGW WenKai, Songti SC, serif">
   817→  <defs>
   818→    ${markers}
   819→  </defs>
   820→  <rect x="0" y="0" width="${w}" height="${h}" fill="${C.bg}"/>
   821→  ${title}
   822→  ${sub}
   823→  ${b.els.join('\n  ')}
   824→</svg>`
   825→}
