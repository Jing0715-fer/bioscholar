    98→function shuffle<T>(arr: readonly T[]): T[] {
    99→  const a = [...arr]
   100→  for (let i = a.length - 1; i > 0; i--) {
   101→    const j = Math.floor(Math.random() * (i + 1))
   102→    const tmp = a[i] as T
   103→    a[i] = a[j] as T
   104→    a[j] = tmp
   105→  }
   106→  return a
   107→}

   109→function answerIndexes(q: QuizQuestion): number[] {
   110→  return Array.isArray(q.answer) ? [...q.answer] : [q.answer]
   111→}
   112→
   113→function judgeAnswer(q: QuizQuestion, user: number[]): boolean {
   114→  if (Array.isArray(q.answer)) {
   115→    if (user.length !== q.answer.length) return false
   116→    const correct = [...q.answer].sort((a, b) => a - b)
   117→    const mine = [...user].sort((a, b) => a - b)
   118→    return correct.every((v, i) => v === mine[i])
   119→  }
   120→  return user.length === 1 && user[0] === q.answer
   121→}
   122→
   123→function formatAnswerText(q: QuizQuestion, indexes: number[]): string {
   124→  if (indexes.length === 0) return '未作答'
   125→  return [...indexes]
   126→    .sort((a, b) => a - b)
   127→    .map((i) => {
   128→      const opt = q.options[i]
   129→      const letter = OPTION_LETTERS[i] ?? '?'
   130→      return opt !== undefined ? `${letter}. ${opt}` : letter
   131→    })
   132→    .join('、')
   133→}

   135→function formatDuration(ms: number): string {
   136→  const total = Math.max(0, Math.floor(ms / 1000))
   137→  const m = Math.floor(total / 60)
   138→  const s = total % 60
   139→  return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`
   140→}

   142→/** 随机挑战抽题：难度均衡（基础 3 · 进阶 5 · 挑战 2，不足从余量补齐） */
   143→function pickRandomQuestions(pool: QuizQuestion[]): QuizQuestion[] {
   144→  if (pool.length <= RANDOM_COUNT) return shuffle(pool)
   145→  const buckets: QuizQuestion[][] = [[], [], []]
   146→  for (const q of pool) {
   147→    buckets[q.difficulty - 1]?.push(q)
   148→  }
   149→  const quotas = [3, 5, 2]
   150→  const picked: QuizQuestion[] = []
   151→  const used = new Set<string>()
   152→  buckets.forEach((bucket, d) => {
   153→    for (const q of shuffle(bucket).slice(0, quotas[d] ?? 0)) {
   154→      picked.push(q)
   155→      used.add(q.id)
   156→    }
   157→  })
   158→  if (picked.length < RANDOM_COUNT) {
   159→    for (const q of shuffle(pool.filter((item) => !used.has(item.id)))) {
   160→      if (picked.length >= RANDOM_COUNT) break
   161→      picked.push(q)
   162→    }
   163→  }
   164→  return shuffle(picked)
   165→}

   190→function ScoreRing({ pct }: { pct: number }) {
   191→  const R = 52
   192→  const CIRC = 2 * Math.PI * R
   193→  const stroke =
   194→    pct >= 80 ? 'stroke-emerald-500' : pct >= 60 ? 'stroke-amber-500' : 'stroke-rose-500'
   195→  const fill =
   196→    pct >= 80
   197→      ? 'fill-emerald-600 dark:fill-emerald-400'
   198→      : pct >= 60
   199→        ? 'fill-amber-600 dark:fill-amber-400'
   200→        : 'fill-rose-600 dark:fill-rose-400'
   201→  return (
   202→    <svg
   203→      viewBox="0 0 128 128"
   204→      className="h-32 w-32 shrink-0"
   205→      role="img"
   206→      aria-label={`得分率 ${pct}%`}
   207→    >
   208→      <circle cx="64" cy="64" r={R} fill="none" className="stroke-muted" strokeWidth="10" />
   209→      <circle
   210→        cx="64"
   211→        cy="64"
   212→        r={R}
   213→        fill="none"
   214→        className={cn(stroke, 'transition-[stroke-dashoffset] duration-700 ease-out')}
   215→        strokeWidth="10"
   216→        strokeLinecap="round"
   217→        strokeDasharray={CIRC}
   218→        strokeDashoffset={CIRC * (1 - Math.min(Math.max(pct, 0), 100) / 100)}
   219→        transform="rotate(-90 64 64)"
   220→      />
   221→      <text
   222→        x="64"
   223→        y="64"
   224→        textAnchor="middle"
   225→        dominantBaseline="middle"
   226→        fontSize="26"
   227→        className={cn(fill, 'font-serif font-bold')}
   228→      >
   229→        {pct}
   230→      </text>
   231→      <text x="64" y="86" textAnchor="middle" fontSize="11" className="fill-muted-foreground">
   232→        得分率 %
   233→      </text>
   234→    </svg>
   235→  )
   236→}

   242→export function QuizView({ subjectId }: { subjectId: SubjectId }) {
   243→  const navigate = useAppStore((s) => s.navigate)
   244→  const theme = getSubjectTheme(subjectId)
   245→  const subject = getSubject(subjectId)
   246→
   247→  const [phase, setPhase] = useState<Phase>('select')
   248→  const [mode, setMode] = useState<Mode>('chapter')
   249→  const [questions, setQuestions] = useState<QuizQuestion[]>([])
   250→  const [activeChapterId, setActiveChapterId] = useState<string | null>(null)
   251→  const [index, setIndex] = useState(0)
   252→  const [answers, setAnswers] = useState<Record<string, number[]>>({})
   253→  const [pending, setPending] = useState<number[]>([])
   254→  const [records, setRecords] = useState<AttemptRecord[]>([])
   255→  const [startedAt, setStartedAt] = useState(0)
   256→  const [elapsed, setElapsed] = useState(0)
   257→  const [now, setNow] = useState(0)
   258→  const [submitting, setSubmitting] = useState(false)
   259→  const [attempts, setAttempts] = useState<AttemptApiItem[]>([])
   260→
   261→  const pool = useMemo(() => getQuizBySubject(subjectId), [subjectId])
   262→  const diffCounts = useMemo(() => {
   263→    const counts = [0, 0, 0]
   264→    for (const q of pool) counts[q.difficulty - 1] += 1
   265→    return counts
   266→  }, [pool])
   267→
   268→  /** 该学科历史答题统计（从 API 全量记录中过滤） */
   269→  const subjectStats = useMemo(() => {
   270→    const mine = attempts.filter((a) => a.subjectId === subjectId)
   271→    const total = mine.length
   272→    const correct = mine.filter((a) => a.correct).length
   273→    return { total, correct, accuracy: total > 0 ? Math.round((correct / total) * 100) : 0 }
   274→  }, [attempts, subjectId])
   275→
   276→  /** 各章节历史答题统计 */
   277→  const chapterStats = useMemo(() => {
   278→    const map = new Map<string, { total: number; correct: number }>()
   279→    for (const a of attempts) {
   280→      if (a.subjectId !== subjectId) continue
   281→      const prev = map.get(a.chapterId)
   282→      if (prev) {
   283→        prev.total += 1
   284→        if (a.correct) prev.correct += 1
   285→      } else {
   286→        map.set(a.chapterId, { total: 1, correct: a.correct ? 1 : 0 })
   287→      }
   288→    }
   289→    return map
   290→  }, [attempts, subjectId])
   291→
   292→  const loadAttempts = useCallback(async () => {
   293→    try {
   294→      const res = await fetch('/api/quiz/attempts')
   295→      if (!res.ok) return
   296→      const data = (await res.json()) as { attempts?: AttemptApiItem[] }
   297→      setAttempts(data.attempts ?? [])
   298→    } catch {
   299→      // 网络异常时静默处理，不影响本地练习
   300→    }
   301→  }, [])
   302→
   303→  useEffect(() => {
   304→    void loadAttempts()
   305→  }, [loadAttempts])
   306→
   307→  // 答题中每秒刷新计时
   308→  useEffect(() => {
   309→    if (phase !== 'quiz') return
   310→    setNow(Date.now())
   311→    const t = setInterval(() => setNow(Date.now()), 1000)
   312→    return () => clearInterval(t)
   313→  }, [phase])
   314→
   315→  const current = questions[index]
   316→  const currentAnswer = current ? (answers[current.id] ?? []) : []
   317→  const answered = current ? answers[current.id] !== undefined : false
   318→  const answeredCount = useMemo(
   319→    () => questions.filter((q) => answers[q.id] !== undefined).length,
   320→    [questions, answers]
   321→  )
   322→
   323→  const startQuiz = useCallback((qs: QuizQuestion[], m: Mode, chapterId: string | null) => {
   324→    setQuestions(qs)
   325→    setMode(m)
   326→    setActiveChapterId(chapterId)
   327→    setIndex(0)
   328→    setAnswers({})
   329→    setPending([])
   330→    setRecords([])
   331→    setStartedAt(Date.now())
   332→    setElapsed(0)
   333→    setPhase('quiz')
   334→  }, [])
   335→
   336→  const startChapter = (chapterId: string) => {
   337→    const qs = getQuizByChapter(chapterId)
   338→    if (qs.length === 0) {
   339→      toast.error('该章节暂无题目')
   340→      return
   341→    }
   342→    startQuiz(shuffle(qs), 'chapter', chapterId)
   343→  }
   344→
   345→  const startRandom = () => {
   346→    if (pool.length === 0) {
   347→      toast.error('该学科暂无题目')
   348→      return
   349→    }
   350→    startQuiz(pickRandomQuestions(pool), 'random', null)
   351→  }
   352→
   353→  const commitAnswer = (userAnswer: number[]) => {
   354→    const q = questions[index]
   355→    if (!q || answers[q.id] !== undefined || userAnswer.length === 0) return
   356→    const correct = judgeAnswer(q, userAnswer)
   357→    setAnswers((prev) => ({ ...prev, [q.id]: userAnswer }))
   358→    setRecords((prev) => [
   359→      ...prev,
   360→      {
   361→        questionId: q.id,
   362→        subjectId: q.subjectId,
   363→        chapterId: q.chapterId,
   364→        userAnswer,
   365→        correct,
   366→      },
   367→    ])
   368→  }
   369→
   370→  const togglePending = (i: number) => {
   371→    setPending((prev) => (prev.includes(i) ? prev.filter((x) => x !== i) : [...prev, i]))
   372→  }
   373→
   374→  const finishQuiz = useCallback(async () => {
   375→    setElapsed(Date.now() - startedAt)
   376→    setPhase('result')
   377→    if (records.length === 0) return
   378→    setSubmitting(true)
   379→    try {
   380→      const res = await fetch('/api/quiz/attempts', {
   381→        method: 'POST',
   382→        headers: { 'Content-Type': 'application/json' },
   383→        body: JSON.stringify({ records }),
   384→      })
   385→      if (!res.ok) throw new Error('submit failed')
   386→      const data = (await res.json()) as { count?: number }
   387→      toast.success(`成绩已记录 · 共提交 ${data.count ?? records.length} 条答题记录`)
   388→      await loadAttempts()
   389→    } catch {
   390→      toast.error('答题记录提交失败，统计可能未更新')
   391→    } finally {
   392→      setSubmitting(false)
   393→    }
   394→  }, [records, startedAt, loadAttempts])
   395→
   396→  const goNext = () => {
   397→    if (!answered) return
   398→    if (index >= questions.length - 1) {
   399→      void finishQuiz()
   400→    } else {
   401→      setIndex((i) => i + 1)
   402→      setPending([])
   403→    }
   404→  }
   405→
   406→  const exitQuiz = () => {
   407→    setPhase('select')
   408→    setQuestions([])
   409→    setPending([])
   410→  }
   411→
   412→  const restart = () => {
   413→    if (mode === 'chapter' && activeChapterId) startChapter(activeChapterId)
   414→    else startRandom()
   415→  }
   416→
   417→  const correctCount = records.filter((r) => r.correct).length
   418→  const scorePct =
   419→    records.length > 0 ? Math.round((correctCount / records.length) * 100) : 0
   420→  const scoreComment =
   421→    scorePct >= 90
   422→      ? '出色！知识点掌握非常扎实'
   423→      : scorePct >= 70
   424→        ? '不错的成绩，再查漏补缺就更完美'
   425→        : scorePct >= 50
   426→          ? '刚过半数，结合解析巩固薄弱环节'
   427→          : '别灰心，读懂解析后再战一轮'
   428→  const scoreTextCls =
   429→    scorePct >= 80
   430→      ? 'text-emerald-600 dark:text-emerald-400'
   431→      : scorePct >= 60
   432→        ? 'text-amber-600 dark:text-amber-400'
   433→        : 'text-rose-600 dark:text-rose-400'
   434→
   435→  const activeChapterTitle =
   436→    mode === 'chapter' && activeChapterId
   437→      ? subject?.chapters.find((c) => c.id === activeChapterId)?.title
   438→      : undefined
   439→
   440→  // ============================================================
   441→  // 渲染
   442→  // ============================================================
   443→
   444→  return (
   445→    <div className="mx-auto max-w-5xl px-4 py-6 sm:px-6 lg:px-8">
   446→      {/* 顶部标题 + 学科统计（编辑式头部） */}
   447→      <header>
   448→        <div className="flex flex-wrap items-end justify-between gap-3">
   449→          <div className="min-w-0">
   450→            <p className="bio-eyebrow text-muted-foreground">Self-Assessment Center</p>
   451→            <h1 className="mt-2 font-serif text-xl font-bold tracking-tight sm:text-2xl">
   452→              测验中心
   453→            </h1>
   454→          </div>
   455→          <p
   456→            className="text-xs text-muted-foreground sm:text-sm"
   457→            aria-label="该学科答题统计"
   458→          >
   459→            <span>已答 </span>
   460→            <span
   461→              className={cn(
   462→                'font-serif text-sm font-bold tabular-nums',
   463→                subjectStats.total > 0 ? theme.classes.text : 'text-foreground'
   464→              )}
   465→            >
   466→              {subjectStats.total}
   467→            </span>
   468→            <span> 题</span>
   469→            <span className="mx-1.5 opacity-30" aria-hidden>
   470→              ·
   471→            </span>
   472→            <span>正确率 </span>
   473→            <span
   474→              className={cn(
   475→                'font-serif text-sm font-bold tabular-nums',
   476→                subjectStats.total > 0 ? theme.classes.text : 'text-foreground'
   477→              )}
   478→            >
   479→              {subjectStats.accuracy}%
   480→            </span>
   481→          </p>
   482→        </div>
   483→        <p className="mt-1.5 text-sm text-muted-foreground">
   484→          {subject ? `${subject.name} · ` : ''}题库共 {pool.length} 题 · 即时判分与解析
   485→        </p>
   486→        <div className="bio-rule mt-4" aria-hidden />
   487→      </header>
   488→
   489→      {/* 学科选择 */}
   490→      <Tabs
   491→        value={subjectId}
   492→        onValueChange={(v) => navigate({ name: 'quiz', subjectId: v as SubjectId })}
   493→        className="mt-4"
   494→      >
   495→        <TabsList className="grid h-auto w-full grid-cols-2 gap-1 sm:grid-cols-4">
   496→          {subjects.map((s) => {
   497→            const t = getSubjectTheme(s.id)
   498→            return (
   499→              <TabsTrigger
   500→                key={s.id}
   501→                value={s.id}
   502→                className={cn('h-9 gap-1.5 hover:bg-accent', SUBJECT_TAB_ACTIVE[s.id])}
   503→              >
   504→                <t.icon className="h-4 w-4" />
   505→                {s.name}
   506→              </TabsTrigger>
   507→            )
   508→          })}
   509→        </TabsList>
   510→      </Tabs>
   511→
   512→      {/* ---------- 模式选择 ---------- */}
   513→      {phase === 'select' && (
   514→        <Tabs
   515→          value={mode}
   516→          onValueChange={(v) => setMode(v as Mode)}
   517→          className="mt-5 bio-fade-up"
   518→        >
   519→          <TabsList>
   520→            <TabsTrigger value="chapter" className="gap-1.5">
   521→              <BookOpen className="h-4 w-4" />
   522→              章节练习
   523→            </TabsTrigger>
   524→            <TabsTrigger value="random" className="gap-1.5">
   525→              <Shuffle className="h-4 w-4" />
   526→              随机挑战
   527→            </TabsTrigger>
   528→          </TabsList>
   529→
   530→          <TabsContent value="chapter" className="mt-4">
   531→            <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
   532→              {(subject?.chapters ?? []).map((ch) => {
   533→                const count = getQuizByChapter(ch.id).length
