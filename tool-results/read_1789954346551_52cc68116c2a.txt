   349→   349→  /* ----- 会话初始化 + 历史加载 ----- */
   350→   350→  useEffect(() => {
   351→   351→    let sid: string | null = null
   352→   352→    try {
   353→   353→      sid = window.localStorage.getItem(SESSION_KEY)
   354→   354→      if (!sid) {
   355→   355→        sid = uuid()
   356→   356→        window.localStorage.setItem(SESSION_KEY, sid)
   357→   357→      }
   358→   358→    } catch {
   359→   359→      // 隐私模式等 localStorage 不可用时退化为内存会话
   360→   360→      sid = uuid()

   362→   362→    sessionIdRef.current = sid
   363→   363→    setSessionId(sid)
   364→   364→
   365→   365→    let cancelled = false
   366→   366→    fetch(`/api/chat?sessionId=${encodeURIComponent(sid)}`)
   367→   367→      .then((r) => {
   368→   368→        if (!r.ok) throw new Error(`HTTP ${r.status}`)
   369→   369→        return r.json() as Promise<unknown>
   370→   370→      })
   371→   371→      .then((data: unknown) => {
   372→   372→        if (cancelled) return
   373→   373→        if (isHistoryResponse(data)) {
   374→   374→          setMessages(data.messages)
   375→   375→          setHadHistory(data.messages.length > 0)
   376→   376→        }
   377→   377→      })
   378→   378→      .catch(() => {
   379→   379→        if (!cancelled) {
   380→   380→          toast({
   381→   381→            title: '加载历史对话失败',
   382→   382→            description: '请刷新页面重试',
   383→   383→            variant: 'destructive',
   384→   384→          })
   385→   385→        }
   386→   386→      })
   387→   387→      .finally(() => {
   388→   388→        if (!cancelled) setHistoryLoaded(true)
   389→   389→      })
   390→   390→
   391→   391→    return () => {
   392→   392→      cancelled = true
   393→   393→    }
   394→   394→  }, [toast])
   395→   395→
   396→   396→  /* ----- 卸载时中止请求 ----- */
   397→   397→  useEffect(() => {
   398→   398→    return () => {
   399→   399→      abortRef.current?.abort()
   400→   400→      if (rafRef.current != null) cancelAnimationFrame(rafRef.current)
   401→   401→    }
   402→   402→  }, [])
   403→   403→
   404→   404→  /* ----- 流式增量 rAF 批量渲染 ----- */
   405→   405→  const flush = useCallback(() => {
   406→   406→    const chunk = pendingRef.current
   407→   407→    pendingRef.current = ''
   408→   408→    const id = streamIdRef.current
   409→   409→    if (!chunk || !id) return
   410→   410→    setMessages((prev) =>
   411→   411→      prev.map((m) => (m.id === id ? { ...m, content: m.content + chunk } : m))
   412→   412→    )
   413→   413→  }, [])
   414→   414→
   415→   415→  const scheduleFlush = useCallback(() => {
   416→   416→    if (rafRef.current == null) {
   417→   417→      rafRef.current = requestAnimationFrame(() => {
   418→   418→        rafRef.current = null
   419→   419→        flush()
   420→   420→      })
   421→   421→    }
   422→   422→  }, [flush])
   423→   423→
   424→   424→  /* ----- 滚动控制：仅当接近底部时自动跟随 ----- */
   425→   425→  const handleScroll = useCallback(() => {
   426→   426→    const el = scrollRef.current
   427→   427→    if (!el) return
   428→   428→    const near = el.scrollHeight - el.scrollTop - el.clientHeight < 80
   429→   429→    stickRef.current = near
   430→   430→    setAtBottom((prev) => (prev === near ? prev : near))
   431→   431→  }, [])
   432→   432→
   433→   433→  const scrollToBottom = useCallback((smooth: boolean) => {
   434→   434→    const el = scrollRef.current
   435→   435→    if (!el) return
   436→   436→    stickRef.current = true
   437→   437→    setAtBottom(true)
   438→   438→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' })
   439→   439→  }, [])
