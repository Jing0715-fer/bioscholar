   117→function isRecord(v: unknown): v is Record<string, unknown> {
   118→  return typeof v === 'object' && v !== null
   119→}
   120→
   121→function isDeltaEvent(v: unknown): v is { delta: string } {
   122→  return isRecord(v) && typeof v.delta === 'string'
   123→}
   124→
   125→function isDoneEvent(v: unknown): v is { done: boolean } {
   126→  return isRecord(v) && typeof v.done === 'boolean'
   127→}
   128→
   129→function isErrorEvent(v: unknown): v is { error: string } {
   130→  return isRecord(v) && typeof v.error === 'string'
   131→}
   132→
   133→function isChatMessage(v: unknown): v is ChatMessage {
   134→  return (
   135→    isRecord(v) &&
   136→    typeof v.id === 'string' &&
   137→    (v.role === 'user' || v.role === 'assistant') &&
   138→    typeof v.content === 'string' &&
   139→    typeof v.createdAt === 'string'
   140→  )
   141→}
   142→
   143→function isHistoryResponse(v: unknown): v is { messages: ChatMessage[] } {
   144→  return isRecord(v) && Array.isArray(v.messages) && v.messages.every(isChatMessage)
   145→}
   146→
   147→function isSubjectId(v: string): v is SubjectId {
   148→  return SUBJECT_IDS.includes(v as SubjectId)
   149→}
   150→
   151→function uuid(): string {
   152→  if (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function') {
   153→    return crypto.randomUUID()
   154→  }
   155→  return `local-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 10)}`
   156→}
   157→
   158→const timeFormatter = new Intl.DateTimeFormat('zh-CN', {
   159→  hour: '2-digit',
   160→  minute: '2-digit',
   161→  hour12: false,
   162→})
   163→
   164→function formatTime(iso: string): string {
   165→  const d = new Date(iso)
   166→  return Number.isNaN(d.getTime()) ? '' : timeFormatter.format(d)
   167→}

   348→export function AssistantView() {
   349→  const assistantContext = useAppStore((s) => s.assistantContext)
   350→  const setAssistantContext = useAppStore((s) => s.setAssistantContext)
   351→  const { toast } = useToast()
   352→
   353→  const [sessionId, setSessionId] = useState<string | null>(null)
   354→  const [historyLoaded, setHistoryLoaded] = useState(false)
   355→  const [hadHistory, setHadHistory] = useState(false)
   356→  const [messages, setMessages] = useState<DisplayMessage[]>([])
   357→  const [input, setInput] = useState('')
   358→  const [generating, setGenerating] = useState(false)
   359→  const [streamingId, setStreamingId] = useState<string | null>(null)
   360→  const [atBottom, setAtBottom] = useState(true)
   361→  const [copiedId, setCopiedId] = useState<string | null>(null)
   362→  const [subjectFilter, setSubjectFilter] = useState<SubjectId | 'all'>('all')
   363→
   364→  const scrollRef = useRef<HTMLDivElement>(null)
   365→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   366→  const abortRef = useRef<AbortController | null>(null)
   367→  const stickRef = useRef(true)
   368→  const rafRef = useRef<number | null>(null)
   369→  const pendingRef = useRef('')
   370→  const streamIdRef = useRef<string | null>(null)
   371→  const sessionIdRef = useRef<string | null>(null)
   372→
   373→  /* ----- 会话初始化 + 历史加载 ----- */
   374→  useEffect(() => {
   375→    let sid: string | null = null
   376→    try {
   377→      sid = window.localStorage.getItem(SESSION_KEY)
   378→      if (!sid) {
   379→        sid = uuid()
   380→        window.localStorage.setItem(SESSION_KEY, sid)
   381→      }
   382→    } catch {
   383→      // 隐私模式等 localStorage 不可用时退化为内存会话
   384→      sid = uuid()
   385→    }
   386→    sessionIdRef.current = sid
   387→    setSessionId(sid)
   388→
   389→    let cancelled = false
   390→    fetch(`/api/chat?sessionId=${encodeURIComponent(sid)}`)
   391→      .then((r) => {
   392→        if (!r.ok) throw new Error(`HTTP ${r.status}`)
   393→        return r.json() as Promise<unknown>
   394→      })
   395→      .then((data: unknown) => {
   396→        if (cancelled) return
   397→        if (isHistoryResponse(data)) {
   398→          setMessages(data.messages)
   399→          setHadHistory(data.messages.length > 0)
   400→        }
   401→      })
   402→      .catch(() => {
   403→        if (!cancelled) {
   404→          toast({
   405→            title: '加载历史对话失败',
   406→            description: '请刷新页面重试',
   407→            variant: 'destructive',
   408→          })
   409→        }
   410→      })
   411→      .finally(() => {
   412→        if (!cancelled) setHistoryLoaded(true)
   413→      })
   414→
   415→    return () => {
   416→      cancelled = true
   417→    }
   418→  }, [toast])
   419→
   420→  /* ----- 卸载时中止请求 ----- */
   421→  useEffect(() => {
   422→    return () => {
   423→      abortRef.current?.abort()
   424→      if (rafRef.current != null) cancelAnimationFrame(rafRef.current)
   425→    }
   426→  }, [])
   427→
   428→  /* ----- 流式增量 rAF 批量渲染 ----- */
   429→  const flush = useCallback(() => {
   430→    const chunk = pendingRef.current
   431→    pendingRef.current = ''
   432→    const id = streamIdRef.current
   433→    if (!chunk || !id) return
   434→    setMessages((prev) =>
   435→      prev.map((m) => (m.id === id ? { ...m, content: m.content + chunk } : m))
   436→    )
   437→  }, [])
   438→
   439→  const scheduleFlush = useCallback(() => {
   440→    if (rafRef.current == null) {
   441→      rafRef.current = requestAnimationFrame(() => {
   442→        rafRef.current = null
   443→        flush()
   444→      })
   445→    }
   446→  }, [flush])
   447→
   448→  /* ----- 滚动控制：仅当接近底部时自动跟随 ----- */
   449→  const handleScroll = useCallback(() => {
   450→    const el = scrollRef.current
   451→    if (!el) return
   452→    const near = el.scrollHeight - el.scrollTop - el.clientHeight < 80
   453→    stickRef.current = near
   454→    setAtBottom((prev) => (prev === near ? prev : near))
   455→  }, [])
   456→
   457→  const scrollToBottom = useCallback((smooth: boolean) => {
   458→    const el = scrollRef.current
   459→    if (!el) return
   460→    stickRef.current = true
   461→    setAtBottom(true)
   462→    el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' })
   463→  }, [])
   464→
   465→  useEffect(() => {
   466→    if (!stickRef.current) return
   467→    const el = scrollRef.current
   468→    if (el) el.scrollTop = el.scrollHeight
   469→  }, [messages])
   470→
   471→  /* ----- 输入框自适应高度 ----- */
   472→  useEffect(() => {
   473→    const el = textareaRef.current
   474→    if (!el) return
   475→    el.style.height = 'auto'
   476→    el.style.height = `${Math.min(el.scrollHeight, MAX_INPUT_HEIGHT)}px`
   477→  }, [input])
   478→
   479→  /* ----- 发送 / 流式接收 ----- */
   480→  const runCompletion = useCallback(
   481→    async (opts: {
   482→      text: string
   483→      context?: ChatContext
   484→      appendUser: boolean
   485→      dropTrailingAssistant?: boolean
   486→    }) => {
   487→      const sid = sessionIdRef.current
   488→      const trimmed = opts.text.trim()
   489→      if (!sid || !trimmed) return
   490→
   491→      const aiId = uuid()
   492→      const now = new Date().toISOString()
   493→      const assistantMsg: DisplayMessage = {
   494→        id: aiId,
   495→        role: 'assistant',
   496→        content: '',
   497→        createdAt: now,
   498→        context: opts.context,
   499→      }
   500→      const userMsg: DisplayMessage = {
   501→        id: uuid(),
   502→        role: 'user',
   503→        content: trimmed,
   504→        createdAt: now,
   505→        context: opts.context,
   506→      }
   507→
   508→      setMessages((prev) => {
   509→        let base = prev
   510→        if (opts.dropTrailingAssistant) {
   511→          while (base.length > 0 && base[base.length - 1].role === 'assistant') {
   512→            base = base.slice(0, -1)
   513→          }
   514→        }
   515→        return opts.appendUser ? [...base, userMsg, assistantMsg] : [...base, assistantMsg]
   516→      })
   517→      setGenerating(true)
   518→      setStreamingId(aiId)
   519→      streamIdRef.current = aiId
   520→      pendingRef.current = ''
   521→      stickRef.current = true
   522→
   523→      const controller = new AbortController()
   524→      abortRef.current = controller
   525→
   526→      let streamError: string | null = null
   527→      let userAborted = false
   528→
   529→      try {
   530→        const res = await fetch('/api/chat', {
   531→          method: 'POST',
   532→          headers: { 'Content-Type': 'application/json' },
   533→          body: JSON.stringify({
   534→            sessionId: sid,
   535→            message: trimmed,
   536→            context: opts.context ?? undefined,
   537→          }),
   538→          signal: controller.signal,
   539→        })
   540→
   541→        if (!res.ok) {
   542→          streamError = `服务响应异常（HTTP ${res.status}）`
   543→        } else {
   544→          const body = res.body
   545→          if (!body) {
   546→            streamError = '响应流不可用'
   547→          } else {
   548→            const reader = body.getReader()
   549→            const decoder = new TextDecoder()
   550→            let buffer = ''
   551→            for (;;) {
   552→              const { done, value } = await reader.read()
   553→              if (done) break
   554→              buffer += decoder.decode(value, { stream: true })
   555→              const lines = buffer.split('\n')
   556→              buffer = lines.pop() ?? ''
   557→              for (const rawLine of lines) {
   558→                const line = rawLine.trim()
   559→                if (!line.startsWith('data:')) continue
   560→                const payload = line.slice(5).trim()
   561→                if (!payload || payload === '[DONE]') continue
   562→                let parsed: unknown
   563→                try {
   564→                  parsed = JSON.parse(payload)
   565→                } catch {
   566→                  continue // 忽略无法解析的行
   567→                }
   568→                if (isErrorEvent(parsed)) {
   569→                  streamError = parsed.error
   570→                  break
   571→                }
   572→                if (isDeltaEvent(parsed)) {
   573→                  pendingRef.current += parsed.delta
   574→                  scheduleFlush()
   575→                }
   576→                // { done: true } 为正常结束标记，无需处理
   577→              }
   578→              if (streamError) break
   579→            }
   580→            // 提前退出时释放底层连接
   581→            void reader.cancel().catch(() => {})
   582→          }
   583→        }
   584→      } catch (e) {
   585→        if (controller.signal.aborted) {
   586→          userAborted = true
   587→        } else if (e instanceof TypeError) {
   588→          streamError = '网络连接失败，请检查网络后重试'
   589→        } else {
   590→          streamError = e instanceof Error ? e.message : '请求失败，请重试'
   591→        }
   592→      } finally {
   593→        if (rafRef.current != null) {
   594→          cancelAnimationFrame(rafRef.current)
   595→          rafRef.current = null
   596→        }
   597→        flush()
   598→        abortRef.current = null
   599→        streamIdRef.current = null
   600→        setStreamingId(null)
   601→        setGenerating(false)
   602→      }
   603→
   604→      if (streamError !== null) {
   605→        const err = streamError
   606→        setMessages((prev) => prev.map((m) => (m.id === aiId ? { ...m, error: err } : m)))
   607→        toast({ title: '回复生成失败', description: err, variant: 'destructive' })
   608→      } else if (userAborted) {
   609→        setMessages((prev) => prev.map((m) => (m.id === aiId ? { ...m, stopped: true } : m)))
   610→      }
   611→    },
   612→    [flush, scheduleFlush, toast]
   613→  )
   614→
   615→  const send = useCallback(
   616→    (raw: string) => {
   617→      const text = raw.trim()
   618→      if (!text || generating || !historyLoaded || sessionId === null) return
   619→      setInput('')
   620→      const ctx: ChatContext | undefined = assistantContext ?? undefined
   621→      textareaRef.current?.focus()
   622→      void runCompletion({ text, context: ctx, appendUser: true })
   623→    },
   624→    [assistantContext, generating, historyLoaded, runCompletion, sessionId]
   625→  )
   626→
   627→  const retryLast = useCallback(() => {
   628→    if (generating || !historyLoaded) return
   629→    const lastUser = [...messages].reverse().find((m) => m.role === 'user')
   630→    if (!lastUser) return
   631→    void runCompletion({
   632→      text: lastUser.content,
   633→      context: lastUser.context ?? undefined,
   634→      appendUser: false,
   635→      dropTrailingAssistant: true,
   636→    })
   637→  }, [generating, historyLoaded, messages, runCompletion])
   638→
   639→  const stop = useCallback(() => {
   640→    abortRef.current?.abort()
   641→  }, [])
   642→
   643→  const clearConversation = useCallback(async () => {
   644→    const sid = sessionIdRef.current
   645→    if (!sid) return
   646→    abortRef.current?.abort()
   647→    try {
   648→      const res = await fetch(`/api/chat?sessionId=${encodeURIComponent(sid)}`, {
   649→        method: 'DELETE',
   650→      })
   651→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   652→      setMessages([])
   653→      setHadHistory(false)
   654→      toast({ title: '对话已清空', description: '已开始新的会话' })
   655→    } catch (e) {
   656→      const desc = e instanceof Error ? e.message : '请稍后重试'
   657→      toast({ title: '清空对话失败', description: desc, variant: 'destructive' })
   658→    }
   659→  }, [toast])
   660→
   661→  const copyMessage = useCallback(
   662→    async (message: DisplayMessage) => {
   663→      try {
   664→        await navigator.clipboard.writeText(message.content)
   665→        setCopiedId(message.id)
   666→        window.setTimeout(() => setCopiedId((cur) => (cur === message.id ? null : cur)), 1600)
   667→      } catch {
   668→        toast({
   669→          title: '复制失败',
   670→          description: '当前浏览器不支持剪贴板操作',
   671→          variant: 'destructive',
   672→        })
   673→      }
   674→    },
   675→    [toast]
   676→  )
   677→
   678→  /* ----- 输入事件 ----- */
   679→  const handleInputChange = useCallback((e: ChangeEvent<HTMLTextAreaElement>) => {
   680→    setInput(e.target.value)
   681→  }, [])
   682→
   683→  const handleKeyDown = useCallback(
   684→    (e: KeyboardEvent<HTMLTextAreaElement>) => {
   685→      if (e.key === 'Enter' && !e.shiftKey && !e.nativeEvent.isComposing) {
   686→        e.preventDefault()
   687→        send(input)
   688→      }
   689→    },
   690→    [input, send]
   691→  )
   692→
   693→  const handleFilterChange = useCallback((v: string) => {
   694→    setSubjectFilter(v === 'all' ? 'all' : isSubjectId(v) ? v : 'all')
   695→  }, [])
   696→
   697→  /* ----- 派生数据 ----- */
   698→  const contextInfo = useMemo(() => {
   699→    if (!assistantContext?.subjectId) return null
   700→    const subject = getSubject(assistantContext.subjectId)
   701→    const chapter = assistantContext.chapterId
   702→      ? getChapter(assistantContext.subjectId, assistantContext.chapterId)
   703→      : undefined
   704→    const parts: string[] = [subject?.name ?? '教材']
   705→    if (chapter) parts.push(`第 ${chapter.number} 章 ${chapter.title}`)
   706→    if (assistantContext.sectionTitle) parts.push(assistantContext.sectionTitle)
   707→    return {
   708→      label: parts.join(' · '),
   709→      theme: subject ? getSubjectTheme(subject.id) : null,
   710→    }
   711→  }, [assistantContext])
   712→
   713→  const filterSubject = useMemo(
   714→    () => (subjectFilter === 'all' ? null : getSubject(subjectFilter)),
   715→    [subjectFilter]
   716→  )
   717→
   718→  const canSend = input.trim().length > 0 && !generating && historyLoaded && sessionId !== null
   719→
   720→  const placeholder = !historyLoaded
   721→    ? '正在加载会话…'
   722→    : filterSubject
   723→      ? `聚焦「${filterSubject.name}」· 输入你的生物学问题…`
   724→      : '输入你的生物学问题，例如：解释糖酵解的反应历程…'
   725→
   726→  /* ----- 渲染 ----- */
   727→  return (
   728→    <div className="mx-auto flex h-[calc(100dvh-10rem)] min-h-[460px] w-full max-w-4xl flex-col px-3 pt-4 sm:px-4 lg:h-[calc(100dvh-8.5rem)]">
   729→      {/* 顶部横幅 */}
   730→      <header className="relative shrink-0 overflow-hidden rounded-t-2xl border border-b-0 bg-gradient-to-r from-emerald-500 via-teal-600 to-teal-700 px-4 py-3.5 text-white sm:px-5 sm:py-4">
   731→        <div className="bio-dna-bg absolute inset-0 opacity-60" aria-hidden />
   732→        <div className="relative flex items-center gap-3">
   733→          <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-white/15 ring-1 ring-white/30 backdrop-blur-sm">
   734→            <Bot className="h-5 w-5" aria-hidden />
   735→          </div>
   736→          <div className="min-w-0 flex-1">
   737→            <h1 className="flex items-center gap-2 text-base font-bold leading-tight sm:text-lg">
   738→              AI 智能助教
   739→              <GraduationCap className="h-4 w-4 shrink-0 opacity-80" aria-hidden />
   740→            </h1>
   741→            <p className="mt-0.5 truncate text-[11px] text-white/80 sm:text-xs">
   742→              基于「101计划」教材体系的生物学专家
   743→            </p>
   744→          </div>
   745→          <span className="hidden shrink-0 items-center gap-1.5 rounded-full bg-white/15 px-2.5 py-1 text-[10px] font-medium ring-1 ring-white/25 sm:inline-flex">
   746→            <span
   747→              className={cn(
   748→                'h-1.5 w-1.5 rounded-full',
   749→                generating ? 'animate-pulse bg-amber-300' : 'bg-emerald-300'
   750→              )}
   751→              aria-hidden
   752→            />
   753→            {generating ? '生成中' : '在线'}
   754→          </span>
   755→        </div>
   756→      </header>
   757→
   758→      {/* 工具栏：教材上下文 / 学科过滤 / 清空 */}
   759→      <div className="flex shrink-0 items-center gap-2 border-x bg-card/60 px-3 py-2 sm:px-4">
   760→        <div className="flex min-w-0 flex-1 items-center">
   761→          {contextInfo ? (
   762→            <Badge
   763→              variant="outline"
   764→              className={cn(
   765→                'gap-1.5 py-1 pl-2.5 pr-1 text-[11px] font-medium',
   766→                contextInfo.theme?.classes.badge ??
   767→                  'bg-emerald-500/15 text-emerald-700 dark:text-emerald-400 border-emerald-500/30'
   768→              )}
   769→            >
   770→              <BookOpenText className="h-3 w-3 shrink-0" aria-hidden />
   771→              <span className="max-w-[13rem] truncate sm:max-w-[24rem]" title={contextInfo.label}>
   772→                {contextInfo.label}
   773→              </span>
   774→              <button
   775→                type="button"
   776→                onClick={() => setAssistantContext(null)}
   777→                aria-label="清除教材上下文"
   778→                className="ml-0.5 rounded-full p-0.5 transition-colors hover:bg-foreground/10"
   779→              >
   780→                <X className="h-3 w-3" aria-hidden />
   781→              </button>
   782→            </Badge>
   783→          ) : filterSubject ? (
   784→            <Badge
   785→              variant="outline"
   786→              className="gap-1.5 border-emerald-500/30 bg-emerald-500/15 py-1 pl-2.5 pr-1 text-[11px] font-medium text-emerald-700 dark:text-emerald-400"
   787→            >
   788→              <Sparkles className="h-3 w-3 shrink-0" aria-hidden />
   789→              <span className="truncate">提问聚焦：{filterSubject.name}</span>
   790→              <button
   791→                type="button"
   792→                onClick={() => setSubjectFilter('all')}
   793→                aria-label="取消学科聚焦"
   794→                className="ml-0.5 rounded-full p-0.5 transition-colors hover:bg-foreground/10"
   795→              >
   796→                <X className="h-3 w-3" aria-hidden />
   797→              </button>
   798→            </Badge>
   799→          ) : (
   800→            <p className="flex min-w-0 items-center gap-1.5 text-[11px] text-muted-foreground">
   801→              <Lightbulb className="h-3.5 w-3.5 shrink-0 text-amber-500" aria-hidden />
   802→              <span className="truncate">
   803→                提示：从知识点阅读页点击「向 AI 提问」可携带教材上下文
   804→              </span>
   805→            </p>
   806→          )}
   807→        </div>
   808→
   809→        <div className="flex shrink-0 items-center gap-1.5">
   810→          <Select value={subjectFilter} onValueChange={handleFilterChange}>
   811→            <SelectTrigger
   812→              size="sm"
   813→              aria-label="选择学科聚焦范围"
   814→              className="h-8 w-[124px] rounded-lg text-xs text-muted-foreground"
   815→            >
   816→              <SelectValue />
   817→            </SelectTrigger>
   818→            <SelectContent>
   819→              <SelectItem value="all">全部学科</SelectItem>
   820→              {subjects.map((s) => (
   821→                <SelectItem key={s.id} value={s.id}>
   822→                  {s.name}
   823→                </SelectItem>
   824→              ))}
   825→            </SelectContent>
   826→          </Select>
   827→
   828→          <AlertDialog>
   829→            <AlertDialogTrigger asChild>
   830→              <Button
   831→                type="button"
   832→                variant="ghost"
   833→                size="icon"
   834→                disabled={messages.length === 0}
   835→                aria-label="清空对话"
   836→                className="h-8 w-8 rounded-lg text-muted-foreground hover:text-destructive"
   837→              >
   838→                <Trash2 className="h-4 w-4" aria-hidden />
   839→              </Button>
   840→            </AlertDialogTrigger>
   841→            <AlertDialogContent className="max-w-sm">
   842→              <AlertDialogHeader>
   843→                <AlertDialogTitle>清空当前对话？</AlertDialogTitle>
   844→                <AlertDialogDescription>
   845→                  将删除本会话的全部 {messages.length} 条消息记录，此操作无法恢复。
   846→                </AlertDialogDescription>
   847→              </AlertDialogHeader>
   848→              <AlertDialogFooter>
   849→                <AlertDialogCancel>取消</AlertDialogCancel>
   850→                <AlertDialogAction
   851→                  onClick={() => void clearConversation()}
   852→                  className="bg-destructive text-white hover:bg-destructive/90"
   853→                >
   854→                  确认清空
   855→                </AlertDialogAction>
   856→              </AlertDialogFooter>
   857→            </AlertDialogContent>
   858→          </AlertDialog>
   859→        </div>
   860→      </div>
   861→
   862→      {/* 消息区 */}
   863→      <div className="relative min-h-0 flex-1 border-x bg-background">
   864→        <div
   865→          ref={scrollRef}
   866→          onScroll={handleScroll}
   867→          role="log"
   868→          aria-label="对话消息"
   869→          aria-live="polite"
   870→          className="bio-scroll h-full overflow-y-auto px-3 py-4 sm:px-5"
   871→        >
   872→          {!historyLoaded ? (
   873→            <HistorySkeleton />
   874→          ) : messages.length === 0 ? (
   875→            <QuickQuestions onPick={send} />
   876→          ) : (
   877→            <div className="flex flex-col gap-5">
   878→              {hadHistory ? (
   879→                <div className="flex items-center gap-3" aria-hidden>
   880→                  <div className="h-px flex-1 bg-border" />
   881→                  <span className="text-[10px] text-muted-foreground">历史消息</span>
   882→                  <div className="h-px flex-1 bg-border" />
   883→                </div>
   884→              ) : null}
   885→              {messages.map((m) => (
   886→                <MessageBubble
   887→                  key={m.id}
   888→                  message={m}
   889→                  streaming={m.id === streamingId}
   890→                  copied={copiedId === m.id}
   891→                  onCopy={copyMessage}
   892→                  onRetry={retryLast}
   893→                />
   894→              ))}
   895→            </div>
   896→          )}
   897→          <div className="h-2" aria-hidden />
   898→        </div>
   899→
   900→        {!atBottom && (
   901→          <button
   902→            type="button"
   903→            onClick={() => scrollToBottom(true)}
   904→            aria-label="滚动到最新消息"
   905→            className="absolute bottom-3 right-3 flex h-9 w-9 items-center justify-center rounded-full border bg-background/90 text-muted-foreground shadow-md backdrop-blur-sm transition-colors hover:text-foreground"
   906→          >
   907→            <ChevronDown className="h-4 w-4" aria-hidden />
   908→          </button>
   909→        )}
   910→      </div>
   911→
   912→      {/* 输入区 */}
   913→      <div className="shrink-0 rounded-b-2xl border border-t bg-card p-2.5 pb-[max(0.625rem,env(safe-area-inset-bottom))] sm:p-3">
   914→        <div className="flex items-end gap-2 rounded-xl border bg-background px-2 py-1 transition-shadow focus-within:border-ring/50 focus-within:ring-2 focus-within:ring-ring/40">
   915→          <Textarea
   916→            ref={textareaRef}
   917→            value={input}
   918→            onChange={handleInputChange}
   919→            onKeyDown={handleKeyDown}
   920→            rows={1}
   921→            placeholder={placeholder}
   922→            aria-label="输入你的问题"
   923→            disabled={!historyLoaded}
   924→            className="min-h-[44px] flex-1 resize-none border-0 bg-transparent px-2 py-2.5 shadow-none focus-visible:border-transparent focus-visible:ring-0 focus-visible:ring-offset-0 dark:bg-transparent"
   925→          />
   926→          {generating ? (
   927→            <Button
   928→              type="button"
   929→              variant="outline"
   930→              size="icon"
   931→              onClick={stop}
   932→              aria-label="停止生成"
   933→              className="h-10 w-10 shrink-0 rounded-lg border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   934→            >
   935→              <Square className="h-4 w-4" aria-hidden />
   936→            </Button>
   937→          ) : (
   938→            <Button
   939→              type="button"
   940→              size="icon"
   941→              onClick={() => send(input)}
   942→              disabled={!canSend}
   943→              aria-label="发送问题"
   944→              className="h-10 w-10 shrink-0 rounded-lg"
   945→            >
   946→              <Send className="h-4 w-4" aria-hidden />
   947→            </Button>
   948→          )}
   949→        </div>
   950→        <p className="mt-1.5 flex flex-wrap items-center gap-x-2 gap-y-0.5 px-1 text-[10px] leading-relaxed text-muted-foreground">
   951→          <span>Enter 发送 · Shift + Enter 换行</span>
   952→          {filterSubject ? <span>当前聚焦「{filterSubject.name}」（提示性，不严格限定）</span> : null}
   953→          <span className="ml-auto hidden sm:inline">回答由 AI 生成，请注意核对关键数据</span>
   954→        </p>
   955→      </div>
   956→    </div>
   957→  )
   958→}
