    44→/** 相邻小节引用（用于上一节/下一节跨章导航） */
    45→interface SectionRef {
    46→  chapter: Chapter
    47→  section: Section
    48→}
    49→
    50→/** 计算上一节/下一节（跨章节边界时自动跳到上一章最后/下一章第一节） */
    51→function computeNeighbors(
    52→  subject: Subject | undefined,
    53→  chapter: Chapter | undefined,
    54→  chapterIndex: number,
    55→  sectionIndex: number
    56→): { prev: SectionRef | null; next: SectionRef | null } {
    57→  if (!subject || !chapter || sectionIndex < 0) {
    58→    return { prev: null, next: null }
    59→  }
    60→  let prev: SectionRef | null = null
    61→  let next: SectionRef | null = null
    62→  if (sectionIndex > 0) {
    63→    prev = { chapter, section: chapter.sections[sectionIndex - 1] }
    64→  } else if (chapterIndex > 0) {
    65→    const pc = subject.chapters[chapterIndex - 1]
    66→    if (pc.sections.length > 0) {
    67→      prev = { chapter: pc, section: pc.sections[pc.sections.length - 1] }
    68→    }
    69→  }
    70→  if (sectionIndex < chapter.sections.length - 1) {
    71→    next = { chapter, section: chapter.sections[sectionIndex + 1] }
    72→  } else if (chapterIndex < subject.chapters.length - 1) {
    73→    const nc = subject.chapters[chapterIndex + 1]
    74→    if (nc.sections.length > 0) {
    75→      next = { chapter: nc, section: nc.sections[0] }
    76→    }
    77→  }
    78→  return { prev, next }
    79→}

   522→function ChapterTree({
   523→  subject,
   524→  currentChapterId,
   525→  currentSectionId,
   526→  completedSections,
   527→  onSelect,
   528→}: {
   529→  subject: Subject
   530→  currentChapterId: string
   531→  currentSectionId: string
   532→  completedSections: string[]
   533→  onSelect: (chapterId: string, sectionId: string) => void
   534→}) {
   535→  const theme = getSubjectTheme(subject.id)
   536→  const [expandedId, setExpandedId] = useState<string | null>(currentChapterId)
   537→
   538→  // 当前章节变化时自动跟随展开
   539→  useEffect(() => {
   540→    setExpandedId(currentChapterId)
   541→  }, [currentChapterId])
   542→
   543→  return (
   544→    <nav aria-label={`${subject.name}章节目录`} className="space-y-1">
   545→      {subject.chapters.map((ch) => {
   546→        const expanded = expandedId === ch.id
   547→        const done = ch.sections.filter((s) => completedSections.includes(s.id)).length
   548→        const allDone = done === ch.sections.length && done > 0
   549→        return (
   550→          <div
   551→            key={ch.id}
   552→            className={cn(
   553→              'rounded-lg transition-colors',
   554→              expanded && theme.classes.bgSoft
   555→            )}
   556→          >
   557→            <button
   558→              onClick={() => setExpandedId(expanded ? null : ch.id)}
   559→              aria-expanded={expanded}
   560→              className="flex w-full items-center gap-2 rounded-lg px-2.5 py-2 text-left text-sm outline-none transition-colors hover:bg-accent focus-visible:ring-2 focus-visible:ring-ring"
   561→            >
   562→              <span
   563→                className={cn(
   564→                  'flex h-6 w-6 shrink-0 items-center justify-center rounded-md text-[11px] font-bold tabular-nums',
   565→                  allDone
   566→                    ? 'bg-primary text-primary-foreground'
   567→                    : cn(theme.classes.bgSoft, theme.classes.text)
   568→                )}
   569→                aria-hidden="true"
   570→              >
   571→                {allDone ? <CheckCircle2 className="h-3.5 w-3.5" /> : ch.number}
   572→              </span>
   573→              <span
   574→                className={cn(
   575→                  'min-w-0 flex-1 truncate',
   576→                  ch.id === currentChapterId ? 'font-semibold' : 'font-medium'
   577→                )}
   578→              >
   579→                {ch.title}
   580→              </span>
   581→              <span className="shrink-0 text-[10px] tabular-nums text-muted-foreground">
   582→                {done}/{ch.sections.length}
   583→              </span>
   584→              <ChevronRight
   585→                className={cn(
   586→                  'h-3.5 w-3.5 shrink-0 text-muted-foreground transition-transform duration-200',
   587→                  expanded && 'rotate-90'
   588→                )}
   589→                aria-hidden="true"
   590→              />
   591→            </button>
   592→            {expanded && (
   593→              <ul className="mt-0.5 space-y-0.5 pb-1.5 pl-3 pr-1.5">
   594→                {ch.sections.map((sec, i) => {
   595→                  const isCurrentSec = sec.id === currentSectionId
   596→                  const secDone = completedSections.includes(sec.id)
   597→                  return (
   598→                    <li key={sec.id}>
   599→                      <button
   600→                        onClick={() => onSelect(ch.id, sec.id)}
   601→                        aria-current={isCurrentSec ? 'page' : undefined}
   602→                        className={cn(
   603→                          'flex w-full items-center gap-2 rounded-md border-l-2 px-2 py-1.5 text-left text-xs transition-colors outline-none focus-visible:ring-2 focus-visible:ring-ring',
   604→                          isCurrentSec
   605→                            ? cn(
   606→                                'font-semibold',
   607→                                theme.classes.border,
   608→                                theme.classes.bgSoft,
   609→                                theme.classes.text
   610→                              )
   611→                            : 'border-transparent text-muted-foreground hover:bg-accent hover:text-foreground'
   612→                        )}
   613→                      >
   614→                        {secDone ? (
   615→                          <CheckCircle2
   616→                            className="h-3.5 w-3.5 shrink-0 text-emerald-500"
   617→                            aria-label="已完成"
   618→                          />
   619→                        ) : (
   620→                          <span
   621→                            className="flex h-3.5 w-3.5 shrink-0 items-center justify-center text-[10px] font-medium tabular-nums text-muted-foreground/60"
   622→                            aria-hidden="true"
   623→                          >
   624→                            {i + 1}
   625→                          </span>
   626→                        )}
   627→                        <span className="min-w-0 flex-1 truncate">
   628→                          {sec.title}
   629→                        </span>
   630→                      </button>
   631→                    </li>
   632→                  )
   633→                })}
   634→              </ul>
   635→            )}
   636→          </div>
   637→        )
   638→      })}
   639→    </nav>
   640→  )
   641→}

   643→// ============================================================
   644→// 右栏：本节要点 / 相关术语 / 学习操作（桌面右栏 + 移动端下方共用）
   645→// ============================================================
   646→function SectionAside({
   647→  subjectId,
   648→  section,
   649→  isCompleted,
   650→  pending,
   651→  onToggleComplete,
   652→  onAskAssistant,
   653→  onOpenGlossary,
   654→}: {
   655→  subjectId: SubjectId
   656→  section: Section
   657→  isCompleted: boolean
   658→  pending: boolean
   659→  onToggleComplete: () => void
   660→  onAskAssistant: () => void
   661→  onOpenGlossary: () => void
   662→}) {
   663→  const theme = getSubjectTheme(subjectId)
   664→
   665→  return (
   666→    <>
   667→      {/* 本节要点 */}
   668→      {section.keyPoints.length > 0 && (
   669→        <Card className={cn('gap-3 py-4', theme.classes.border)}>
   670→          <CardHeader className="px-4 pb-0">
   671→            <CardTitle
   672→              className={cn('flex items-center gap-2 text-sm', theme.classes.text)}
   673→            >
   674→              <Lightbulb className="h-4 w-4 shrink-0" aria-hidden="true" />
   675→              本节要点
   676→            </CardTitle>
   677→          </CardHeader>
   678→          <CardContent className="px-4">
   679→            <ul className="space-y-2.5">
   680→              {section.keyPoints.map((kp, i) => (
   681→                <li key={i} className="flex gap-2.5 text-xs leading-relaxed">
   682→                  <span
   683→                    className={cn(
   684→                      'mt-px flex h-5 w-5 shrink-0 items-center justify-center rounded-full text-[10px] font-bold tabular-nums',
   685→                      theme.classes.bgSoft,
   686→                      theme.classes.text
   687→                    )}
   688→                    aria-hidden="true"
   689→                  >
   690→                    {i + 1}
   691→                  </span>
   692→                  <span className="min-w-0 flex-1">{kp}</span>
   693→                </li>
   694→              ))}
   695→            </ul>
   696→          </CardContent>
   697→        </Card>
   698→      )}
   699→
   700→      {/* 相关术语 */}
   701→      {section.terms.length > 0 && (
   702→        <Card className="gap-3 py-4">
   703→          <CardHeader className="px-4 pb-0">
   704→            <CardTitle className="flex items-center gap-2 text-sm">
   705→              <BookMarked
   706→                className="h-4 w-4 shrink-0 text-primary"
   707→                aria-hidden="true"
   708→              />
   709→              相关术语
   710→            </CardTitle>
   711→          </CardHeader>
   712→          <CardContent className="px-4">
   713→            <div className="flex flex-wrap gap-1.5">
   714→              {section.terms.map((term) => (
   715→                <button
   716→                  key={term}
   717→                  onClick={onOpenGlossary}
   718→                  className={cn(
   719→                    'inline-flex items-center gap-1 rounded-md border px-2 py-1 text-xs transition-all outline-none hover:-translate-y-px focus-visible:ring-2 focus-visible:ring-ring',
   720→                    theme.classes.badge
   721→                  )}
   722→                  aria-label={`在术语词典中查看「${term}」`}
   723→                >
   724→                  <BookMarked className="h-3 w-3 opacity-60" aria-hidden="true" />
   725→                  {term}
   726→                </button>
   727→              ))}
   728→            </div>
   729→            <p className="mt-2.5 text-[10px] text-muted-foreground">
   730→              点击术语可跳转术语词典查看详细释义
   731→            </p>
   732→          </CardContent>
   733→        </Card>
   734→      )}
   735→
   736→      {/* 学习操作 */}
   737→      <Card className="gap-3 py-4">
   738→        <CardContent className="space-y-3 px-4">
   739→          {isCompleted && (
   740→            <p className="flex items-center justify-center gap-1.5 text-xs font-medium text-emerald-600 dark:text-emerald-400">
   741→              <CheckCircle2 className="h-4 w-4" aria-hidden="true" />
   742→              已完成本节学习
   743→            </p>
   744→          )}
   745→          {isCompleted ? (
   746→            <Button
   747→              variant="outline"
   748→              className="w-full gap-2"
   749→              onClick={onToggleComplete}
   750→              disabled={pending}
   751→            >
   752→              {pending ? (
   753→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   754→              ) : (
   755→                <Undo2 className="h-4 w-4" aria-hidden="true" />
   756→              )}
   757→              取消已学
   758→            </Button>
   759→          ) : (
   760→            <Button
   761→              className="w-full gap-2 bg-emerald-600 text-white hover:bg-emerald-700"
   762→              onClick={onToggleComplete}
   763→              disabled={pending}
   764→            >
   765→              {pending ? (
   766→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" />
   767→              ) : (
   768→                <CheckCircle2 className="h-4 w-4" aria-hidden="true" />
   769→              )}
   770→              标记为已学
   771→            </Button>
   772→          )}
   773→          <Separator />
   774→          <Tooltip>
   775→            <TooltipTrigger asChild>
   776→              <Button
   777→                variant="outline"
   778→                className={cn('w-full gap-2', theme.classes.hover)}
   779→                onClick={onAskAssistant}
   780→              >
   781→                <MessageSquareText
   782→                  className="h-4 w-4"
   783→                  aria-hidden="true"
   784→                />
   785→                向 AI 助教提问本节
   786→              </Button>
   787→            </TooltipTrigger>
   788→            <TooltipContent>将携带本节学习上下文进入对话</TooltipContent>
   789→          </Tooltip>
   790→        </CardContent>
   791→      </Card>
   792→    </>
   793→  )
   794→}

   799→function SectionNavCard({
   800→  direction,
   801→  target,
   802→  currentChapterId,
   803→  hover,
   804→  onClick,
   805→}: {
   806→  direction: 'prev' | 'next'
   807→  target: SectionRef
   808→  currentChapterId: string
   809→  hover: string
   810→  onClick: () => void
   811→}) {
   812→  const isNext = direction === 'next'
   813→  const crossChapter = target.chapter.id !== currentChapterId
   814→
   815→  return (
   816→    <button
   817→      onClick={onClick}
   818→      className={cn(
   819→        'group flex flex-1 items-center gap-3 rounded-xl border bg-card p-4 text-left outline-none transition-all duration-200 hover:-translate-y-0.5 hover:shadow-md focus-visible:ring-2 focus-visible:ring-ring',
   820→        hover
   821→      )}
   822→      aria-label={`${isNext ? '下一节' : '上一节'}：第 ${target.chapter.number} 章 ${target.section.title}`}
   823→    >
   824→      {!isNext && (
   825→        <span
   826→          className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-muted transition-colors group-hover:bg-accent"
   827→          aria-hidden="true"
   828→        >
   829→          <ArrowLeft className="h-4 w-4" />
   830→        </span>
   831→      )}

   833→        <span
   834→          className={cn(
   835→            'flex items-center gap-1.5 text-[11px] text-muted-foreground',
   836→            isNext && 'sm:justify-end'
   837→          )}
   838→        >
   839→          {crossChapter && (
   840→            <Badge variant="outline" className="h-4 px-1.5 text-[10px]">
   841→              {isNext ? '下一章' : '上一章'}

   843→          )}

   854→        <span
   855→          className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-muted transition-colors group-hover:bg-accent"

   858→          <ArrowRight className="h-4 w-4 transition-transform group-hover:translate-x-0.5" />
   859→        </span>
   860→      )}
   861→    </button>
   862→  )
